import fs from 'node:fs';
import path from 'node:path';
import { createHash } from 'node:crypto';
import { execFileSync } from 'node:child_process';
import { checkBuild } from '../../scripts/start-lib.mjs';

const root = process.cwd();
const version = JSON.parse(fs.readFileSync('packages/ts-shell/package.json', 'utf8')).version;
const checked = checkBuild({ root, dist: path.join(root, 'packages/ts-shell/dist'), version });
if (!checked.ok) throw new Error(checked.reason);
const hashes = JSON.parse(fs.readFileSync('docs/architecture/hooks/evidence/stop-envelope/source-files.json', 'utf8'));
const drift = Object.entries(hashes).filter(([file, hash]) => !fs.existsSync(file)
  || createHash('sha256').update(fs.readFileSync(file)).digest('hex') !== hash);
if (drift.length) throw new Error(`Source drift: ${drift.map(([file]) => file).join(', ')}`);
const reference = '/home/wang/projects/normproof';
const referenceStatus = execFileSync('git', ['-C', reference, 'status', '--porcelain'], { encoding: 'utf8' }).trim();
if (referenceStatus) throw new Error('Reference worktree is no longer clean');
console.log(JSON.stringify({ buildFresh: true, version, sourceFiles: Object.keys(hashes).length,
  sourceDrift: drift.length, referenceClean: true,
  referenceHead: execFileSync('git', ['-C', reference, 'rev-parse', 'HEAD'], { encoding: 'utf8' }).trim() }));
