import fs from 'node:fs';
import path from 'node:path';
import http from 'node:http';
import assert from 'node:assert/strict';
import { spawn } from 'node:child_process';
import { createHash } from 'node:crypto';
import { fileURLToPath } from 'node:url';

const directory = path.dirname(fileURLToPath(import.meta.url)), repo = path.resolve(directory, '../..');
const manifest = JSON.parse(fs.readFileSync(path.join(repo, 'packages/ts-shell/dist/build-manifest.json')));
const receipt = { platform: process.platform, buildId: manifest.buildId, probeSha256: createHash('sha256')
  .update(fs.readFileSync(fileURLToPath(import.meta.url))).digest('hex'), status: 'not_run', stages: {}, posts: 0 };
const root = fs.mkdtempSync(path.join(directory, 'native-state-'));
const clients = new Set(), bodies = [];
let stage = 'setup', server, baseUrl, number = 0;
const delay = ms => new Promise(resolve => setTimeout(resolve, ms));
const watchdog = setTimeout(() => { for (const client of clients) client.process.kill('SIGTERM'); }, 150000);

function setup() {
  const home = path.join(root, 'home'), work = path.join(root, 'work'), storage = path.join(home, '.cheers');
  fs.mkdirSync(storage, { recursive: true }); fs.mkdirSync(work);
  fs.writeFileSync(path.join(storage, 'config.json'), JSON.stringify({ maxOutputTokens: 128,
    compactionEnabled: false, retryEnabled: false, cacheWarming: 'off', reserveTokens: 0, keepRecentTokens: 80 }));
  fs.writeFileSync(path.join(storage, 'auth.json'), JSON.stringify({ provider: 'openai', providerCredentials: {
    openai: { model: 'partial', baseUrl, apiKey: 'synthetic-settings', thinkingEffort: null } } }), { mode: 0o600 });
  fs.writeFileSync(path.join(storage, 'models.json'), JSON.stringify({ providers: { openai: {
    runtime: 'openai', endpoint: 'openai-compatible', auth: 'api-key', baseUrl, models: [
      { id: 'partial', contextWindow: 64000, maxTokens: 2048, reasoning: true,
        thinkingLevelMap: { minimal: null, low: null, medium: null, high: 'high', xhigh: null, max: 'max' } },
      { id: 'plain', contextWindow: 64000, maxTokens: 2048, reasoning: false },
    ] } } }));
  const extension = path.join(work, 'commands.ts');
  fs.writeFileSync(extension, `export default api => {
    api.registerCommand({ name: 'acknowledge', kind: 'local', summary: 'Acknowledge the command', group: 'session',
      aliases: ['ack'], supportsNonInteractive: true, execute: async () => ({ kind: 'local', output: 'RPC_EXTENSION_OK' }) });
    api.registerVirtualModel({ provider: 'adaptive', id: 'auto', name: 'Adaptive', thinkingLevels: ['high', 'max'],
      route: () => ({ provider: 'openai', modelId: 'partial' }) });
  };`);
  const skill = path.join(work, 'explicit', 'SKILL.md'); fs.mkdirSync(path.dirname(skill), { recursive: true });
  fs.writeFileSync(skill, '---\nname: explicit-check\ndescription: Explicit probe instructions\n---\nReturn RPC_SKILL_OK.\n');
  const discovered = path.join(storage, 'skills', 'hidden-check', 'SKILL.md'); fs.mkdirSync(path.dirname(discovered), { recursive: true });
  fs.writeFileSync(discovered, '---\nname: hidden-check\ndescription: Discovered probe instructions\n---\nReturn RPC_DISCOVERED_OK.\n');
  const input = path.join(work, 'overlay.json'); fs.writeFileSync(input, '{"retryEnabled":false}');
  return { home, work, storage, extension, skill, input };
}

function start(fixture, extra = []) {
  const frames = [], waiters = new Set(); let buffer = '', invalid = 0;
  const env = { ...process.env, HOME: fixture.home, USERPROFILE: fixture.home, CHEERS_TEST_ROOT: fixture.home,
    GIT_CEILING_DIRECTORIES: root, CHEERS_PROVIDER_MODE: 'real', CHEERS_REAL_PROVIDER_SMOKE: '', OPENAI_API_KEY: '',
    ANTHROPIC_API_KEY: '', CHEERS_MARKETPLACES_FILE: path.join(fixture.home, 'sources.json'),
    CHEERS_EXTENSIONS_FILE: path.join(fixture.home, 'extensions.json') };
  const spawned = spawn('rtk', ['node', path.join(repo, 'scripts/start.mjs'), '--rpc', '--project', fixture.work,
    '--no-context-files', '--no-skills', '--no-tools', '--extension', fixture.extension, '--skill', fixture.skill, ...extra],
    { cwd: fixture.work, env, stdio: ['pipe', 'pipe', 'pipe'] });
  const closed = new Promise(resolve => spawned.on('close', (code, signal) => resolve({ code, signal })));
  spawned.stderr.on('data', () => {});
  spawned.stdout.on('data', chunk => {
    buffer += chunk.toString();
    for (;;) {
      const end = buffer.indexOf('\n'); if (end < 0) break;
      const line = buffer.slice(0, end); buffer = buffer.slice(end + 1); if (!line.trim()) continue;
      let value; try { value = JSON.parse(line); } catch { invalid++; continue; }
      frames.push(value);
      for (const waiter of [...waiters]) if (waiter.match(value)) waiter.resolve(value);
    }
  });
  function waitFor(match, from = frames.length) {
    const ready = frames.slice(from).find(match); if (ready) return Promise.resolve(ready);
    return new Promise((resolve, reject) => {
      const timer = setTimeout(() => { waiters.delete(waiter); reject(new Error('Response deadline')); }, 25000);
      const waiter = { match, resolve: value => { clearTimeout(timer); waiters.delete(waiter); resolve(value); } };
      waiters.add(waiter);
    });
  }
  const client = { ...fixture, process: spawned, closed, frames, waitFor,
    command(type, args = {}) { const id = String(++number); const result = waitFor(value => value.type === 'response' && value.id === id);
      spawned.stdin.write(JSON.stringify({ id, type, ...args }) + '\n'); return result; },
    async finish() { spawned.stdin.end(); assert.deepEqual(await closed, { code: 0, signal: null });
      clients.delete(client); assert.equal(invalid, 0); assert.equal(buffer.trim(), ''); },
  };
  clients.add(client); return client;
}

try {
  assert.equal(process.platform, 'linux');
  server = http.createServer(async (request, response) => {
    const chunks = []; for await (const chunk of request) chunks.push(Buffer.from(chunk));
    if (request.method !== 'POST') { response.writeHead(200, { 'content-type': 'application/json' }); response.end('{"data":[]}'); return; }
    assert.equal(request.headers.authorization, 'Bearer synthetic-settings');
    const body = JSON.parse(Buffer.concat(chunks).toString()); bodies.push(body); receipt.posts++;
    response.writeHead(200, { 'content-type': 'application/json' });
    response.end(JSON.stringify({ choices: [{ message: { role: 'assistant', content: 'RPC_SETTINGS_OK' }, finish_reason: 'stop' }] }));
  });
  await new Promise(resolve => server.listen(0, '127.0.0.1', resolve)); baseUrl = `http://127.0.0.1:${server.address().port}/v1`;
  const fixture = setup(); let client = start(fixture);
  stage = 'thinking';
  const initial = await client.command('get_state'); assert.equal(initial.success, true); const id = initial.data.sessionId;
  const available = await client.command('get_available_thinking_levels');
  assert.deepEqual(available.data.levels, ['high', 'max', 'off']); assert.deepEqual(available.data.unverifiedLevels, []);
  for (const level of ['high', 'max', 'auto', 'high']) {
    const cycled = await client.command('cycle_thinking_level'); assert.equal(cycled.success, true); assert.equal(cycled.data.level, level);
    assert.equal((await client.command('get_state')).data.thinkingLevel, level);
  }
  const terminal = client.waitFor(value => value.type === 'agent_end');
  assert.equal((await client.command('prompt', { message: 'Return RPC_SETTINGS_OK.' })).success, true);
  assert.equal((await terminal).status, 'completed'); assert.equal(bodies.at(-1).reasoning_effort, 'high');
  assert.equal((await client.command('set_thinking_level', { level: 'auto' })).data.level, 'auto');
  assert.equal((await client.command('set_model', { provider: 'openai', modelId: 'plain' })).success, true);
  assert.deepEqual((await client.command('get_available_thinking_levels')).data.levels, ['off']);
  assert.equal((await client.command('cycle_thinking_level')).data.level, null);
  assert.equal((await client.command('set_model', { provider: 'openai', modelId: 'partial' })).success, true);
  receipt.stages.thinking = { passed: true, actualWireEffort: 'high', sequence: ['high', 'max', 'auto', 'high'], noOpWhenDisabled: true };
  stage = 'settings';
  for (const type of ['set_auto_compaction', 'set_auto_retry']) {
    const result = await client.command(type, { enabled: true }); assert.equal(result.success, true);
    assert.equal(result.data.saved, true); assert.equal(result.data.stored, true); assert.equal(result.data.effective, true);
    assert.equal(result.data.applied, true); assert.equal(result.data.application, 'next operation');
  }
  stage = 'discovery';
  const discovery = await client.command('get_commands'); assert.equal(discovery.success, true);
  const commands = discovery.data.commands, names = commands.map(row => row.name);
  for (const name of ['acknowledge', 'ack', 'reload-plugins', 'recall', 'retain', 'join', 'leave', 'skill:explicit-check']) assert.ok(names.includes(name), name);
  assert.equal(names.includes('skill:hidden-check'), false); assert.equal(new Set(names).size, names.length);
  const skillEnd = client.waitFor(value => value.type === 'agent_end');
  assert.equal((await client.command('prompt', { message: '/skill:explicit-check Check instructions.' })).success, true);
  assert.equal((await skillEnd).status, 'completed'); assert.ok(JSON.stringify(bodies.at(-1)).includes('Return RPC_SKILL_OK.'));
  receipt.stages.discovery = { passed: true, extensionAndAlias: true, namedInvocations: true, explicitSkillUnderNoSkills: true,
    discoveredSkillsSuppressed: true, skillReachedWire: true, uniqueNames: true };
  stage = 'virtual';
  assert.equal((await client.command('set_model', { provider: 'adaptive', modelId: 'auto' })).success, true);
  assert.deepEqual((await client.command('get_available_thinking_levels')).data.levels, ['high', 'max', 'off']);
  assert.equal((await client.command('cycle_thinking_level')).data.level, 'high');
  await client.finish(); client = start(fixture, ['--session-id', id]);
  const reopened = await client.command('get_state'); assert.equal(reopened.data.sessionId, id);
  assert.deepEqual(reopened.data.model, { provider: 'adaptive', id: 'auto' }); assert.equal(reopened.data.thinkingLevel, 'high');
  assert.equal(reopened.data.autoCompactionEnabled, true); assert.equal(reopened.data.autoRetryEnabled, true);
  assert.equal((await client.command('cycle_thinking_level')).data.level, 'max');
  receipt.stages.persistence = { passed: true, sameSession: true, virtualLevel: true, defaults: true, eofExit0: true };
  await client.finish();
  stage = 'precedence'; client = start(fixture, ['--session-id', id, '--config', fixture.input]);
  const selected = await client.command('get_state'); assert.equal(selected.data.autoRetryEnabled, false);
  const before = fs.readFileSync(fixture.input);
  const saved = await client.command('set_auto_retry', { enabled: true }); assert.equal(saved.success, true);
  assert.equal(saved.data.saved, true); assert.equal(saved.data.stored, true); assert.equal(saved.data.effective, false);
  assert.equal(saved.data.applied, false); assert.ok(saved.data.source.includes('input 1'));
  assert.equal((await client.command('get_state')).data.autoRetryEnabled, false); assert.deepEqual(fs.readFileSync(fixture.input), before);
  await client.finish(); receipt.stages.precedence = { passed: true, explicitInputPreserved: true, storedTrueEffectiveFalse: true };
  receipt.status = 'passed';
} catch (error) {
  receipt.status = 'failed'; receipt.failedStage = stage; receipt.failure = { name: error?.name, code: error?.code,
    message: error?.code === 'ERR_ASSERTION' ? 'Assertion failed; see stage.' : 'Native probe failed; details withheld.' };
  process.exitCode = 1;
} finally {
  clearTimeout(watchdog);
  for (const client of clients) { client.process.kill('SIGTERM'); await Promise.race([client.closed, delay(3000)]); }
  server?.closeAllConnections?.(); if (server) await new Promise(resolve => server.close(resolve));
  fs.writeFileSync(path.join(directory, 'settings.json'), JSON.stringify(receipt, null, 2) + '\n');
  console.log(JSON.stringify(receipt)); fs.rmSync(root, { recursive: true, force: true });
}
