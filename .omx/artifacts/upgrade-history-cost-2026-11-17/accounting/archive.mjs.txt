import fs from 'node:fs';
import path from 'node:path';
import { createHash } from 'node:crypto';
import { spawnSync } from 'node:child_process';
import { buildInputDigest } from '../../scripts/build-identity.mjs';

const root = '/home/wang/projects/cheers';
const scratch = path.join(root, '.tmp/history-cost-validation');
const output = path.join(root, '.omx/artifacts/upgrade-history-cost-2026-11-17');
const hash = bytes => createHash('sha256').update(bytes).digest('hex');
const read = file => JSON.parse(fs.readFileSync(path.join(root, file), 'utf8'));
const source = read('.tmp/history-cost-validation/current-sources.json').source;
const manifest = read('.tmp/history-cost-validation/build-manifest.json');
const mismatches = Object.entries(source).filter(([file, digest]) => hash(fs.readFileSync(path.join(root, file))) !== digest);
const inputs = buildInputDigest(root, manifest.compilerInputs);
if (mismatches.length || inputs.digest !== manifest.inputsDigest) throw new Error('Frozen source/build identity changed');
fs.mkdirSync(output, { recursive: true });

function copyFile(file, target) {
  const bytes = fs.readFileSync(file);
  const destination = path.join(output, target);
  fs.mkdirSync(path.dirname(destination), { recursive: true });
  fs.writeFileSync(destination, bytes);
  return { snapshot: target, sha256: hash(bytes) };
}

function copyDirectory(directory, prefix) {
  for (const entry of fs.readdirSync(directory, { withFileTypes: true })) {
    const file = path.join(directory, entry.name);
    const target = path.join(prefix, entry.name);
    if (entry.isDirectory()) copyDirectory(file, target);
    else if (entry.isFile()) copyFile(file, /\.(?:mjs|py|ts)$/.test(entry.name) ? target + '.txt' : target);
  }
}

for (const [directory, prefix] of [
  [scratch, 'accounting'],
  [path.join(root, '.tmp/continuation-validation'), 'recovery'],
]) {
  for (const entry of fs.readdirSync(directory, { withFileTypes: true })) {
    if (!entry.isFile()) continue;
    const file = path.join(directory, entry.name);
    copyFile(file, path.join(prefix, /\.(?:mjs|py|ts)$/.test(entry.name) ? entry.name + '.txt' : entry.name));
  }
}
copyDirectory(path.join(root, '.normproof/out'), 'normproof/normal-judge');
const snapshots = {};

  'cli/interactive/picker/model.test.ts'];

  { cwd: root, encoding: 'utf8', maxBuffer: 2 * 1024 * 1024 });
