import fs from 'node:fs';
import path from 'node:path';
import { createHash } from 'node:crypto';
import { spawnSync } from 'node:child_process';

const root = '/home/wang/projects/cheers';
const tree = path.join(root, '.tmp/continuation-reversal');
const output = path.join(root, '.tmp/ui-renewal-validation');
const fields = ['sdk/agentSession.ts', 'sdk/ui.ts', 'sdk/ui.test.ts', 'sdk/ui/retirement.test.ts', 'config/state.ts'];
const originals = new Map(fields.map(file => [file, fs.readFileSync(path.join(root, 'packages/ts-shell/src', file))]));
const hash = bytes => createHash('sha256').update(bytes).digest('hex');
for (const [file, bytes] of originals) {
  const target = path.join(tree, 'packages/ts-shell/src', file);
  fs.mkdirSync(path.dirname(target), { recursive: true });
  fs.writeFileSync(target, bytes);
}
const cases = [
  { name: 'fresh-sdk-call-renewal', file: 'sdk/ui.ts', from: '      this.install(retained);', to: '      return this.snapshot;',
    test: 'src/sdk/ui.test.ts', filter: 'renews retained', failures: 2 },
  { name: 'unobserved-blocked-publication', file: 'config/state.ts', from: '    if (block) this.revocationGeneration++;',
    to: '    // Reversed for the isolated revocation counterexample.', test: 'src/sdk/ui/retirement.test.ts',
    filter: 'requires an explicit new binding', failures: 3 },
];
const receipt = { status: 'running', mainSourcesMutated: false, cases: [] };
try {
  for (const item of cases) {
    const file = path.join(tree, 'packages/ts-shell/src', item.file);
    const source = originals.get(item.file).toString('utf8');
    if (source.split(item.from).length !== 2) throw new Error('Reversal fragment not unique: ' + item.name);
    fs.writeFileSync(file, source.replace(item.from, item.to));
    const reportFile = path.join(output, 'reversal-' + item.name + '.json');
    const run = spawnSync('rtk', ['env', 'NODE_OPTIONS=--max-old-space-size=1536', 'VITEST_MAX_THREADS=1', 'VITEST_MIN_THREADS=1',
      'GIT_CEILING_DIRECTORIES=' + path.dirname(tree), 'npm', 'exec', '-w', 'packages/ts-shell', '--', 'vitest', 'run',
      item.test, '-t', item.filter, '--no-file-parallelism', '--maxWorkers=1', '--minWorkers=1', '--reporter=json', '--outputFile=' + reportFile],
      { cwd: tree, encoding: 'utf8', timeout: 90000, maxBuffer: 1024 * 1024 });
    fs.writeFileSync(path.join(output, 'reversal-' + item.name + '.log'), (run.stdout ?? '') + (run.stderr ?? ''));
    const report = JSON.parse(fs.readFileSync(reportFile, 'utf8'));
    const failed = report.testResults.flatMap(suite => suite.assertionResults.filter(test => test.status === 'failed'));
    const assertions = failed.filter(test => test.failureMessages.some(message => message.includes('AssertionError'))).length;
    fs.writeFileSync(file, originals.get(item.file));
    receipt.cases.push({ name: item.name, processExitCode: run.status, selected: report.numPassedTests + report.numFailedTests,
      assertionFailures: assertions, passedControls: report.numPassedTests, restoredExactly: hash(fs.readFileSync(file)) === hash(originals.get(item.file)) });
    if (run.status !== 1 || assertions !== item.failures) throw new Error('Ineffective reversal: ' + item.name);
  }
  receipt.status = 'passed';
} catch (error) {
  receipt.status = 'failed'; receipt.failure = String(error.message); process.exitCode = 1;
} finally {
  for (const [file, bytes] of originals) fs.writeFileSync(path.join(tree, 'packages/ts-shell/src', file), bytes);
  receipt.restoredExactly = fields.every(file => hash(fs.readFileSync(path.join(tree, 'packages/ts-shell/src', file))) === hash(originals.get(file)));
  receipt.mainSourcesMatch = fields.every(file => hash(fs.readFileSync(path.join(root, 'packages/ts-shell/src', file))) === hash(originals.get(file)));
  fs.writeFileSync(path.join(output, 'reversals.json'), JSON.stringify(receipt, null, 2) + '\n');
  console.log(JSON.stringify(receipt));
}
