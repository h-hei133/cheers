async function run(code, args, cwd, timeout = 180_000) {
  stage = code;
  const child = spawn('rtk', args, { cwd, env: { ...process.env }, stdio: ['ignore', 'pipe', 'pipe'] });
  let stdout = '', stderr = '', bytes = 0, timedOut = false, overflow = false;
  const timer = setTimeout(() => { timedOut = true; child.kill('SIGKILL'); }, timeout);
  const collect = (chunk, output) => {
    bytes += chunk.length;
    if (bytes > 8 * 1024 * 1024) { overflow = true; child.kill('SIGKILL'); return; }
    if (output) stdout += chunk.toString('utf8'); else stderr += chunk.toString('utf8');
  };
  child.stdout.on('data', chunk => collect(chunk, true)); child.stderr.on('data', chunk => collect(chunk, false));
  try {
    const result = await new Promise(resolve => {
      child.once('error', () => resolve({ exit: null, signal: 'spawn-error' }));
      child.once('close', (exit, signal) => resolve({ exit, signal }));
    });
    fs.writeFileSync(path.join(directory, 'package-' + code + '.log'), stdout + stderr, {mode:0o600});
    receipt.commands.push({ code, ...result, timedOut, overflow, bytes, stdoutSha256: hash(stdout), stderrSha256: hash(stderr) });
    check(result.exit === 0 && !result.signal && !timedOut && !overflow, code + '-failed');
    return stdout;
  } finally { clearTimeout(timer); }
}
