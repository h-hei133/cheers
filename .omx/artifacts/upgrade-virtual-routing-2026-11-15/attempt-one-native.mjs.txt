async function wire(label) {
  const requests = [];
  const server = http.createServer(async (request, response) => {
    try {
      let bytes = 0; const chunks = [];
      for await (const chunk of request) { bytes += chunk.length; if (bytes > 8 * 1024 * 1024) throw new Error('Fixture request exceeds limit.'); chunks.push(Buffer.from(chunk)); }
      if (request.method !== 'POST') {
        response.writeHead(200, { 'content-type': 'application/json' }); response.end(JSON.stringify({ data: [{ id: label }] })); return;
      }
      const body = JSON.parse(Buffer.concat(chunks).toString('utf8'));
      requests.push({ path: request.url, keyMatches: request.headers.authorization === `Bearer synthetic-${label}`, body });
      response.writeHead(200, { 'content-type': 'application/json' });
      response.end(JSON.stringify({ choices: [{ message: { role: 'assistant', content: 'PHYSICAL_ACK' }, finish_reason: 'stop' }],
        usage: { prompt_tokens: 10, completion_tokens: 3, total_tokens: 13 } }));
    } catch { response.writeHead(400); response.end('Invalid fixture request.'); }
  });
  servers.add(server); await new Promise(resolve => server.listen(0, '127.0.0.1', resolve));
  return { requests, baseUrl: `http://127.0.0.1:${server.address().port}/v1` };
}

function routingEvents(session) {
  return session.sessionManager.getTranscriptEvents(session.sessionId).filter(event => event.type === 'branch-store' && event.namespace === 'model-routing');
}
function routeState(session, modelId) {
  const key = JSON.stringify(['adaptive', modelId]);
  const events = routingEvents(session);
  for (let index = events.length - 1; index >= 0; index--) {
    if (Object.hasOwn(events[index].entries, key)) {
      const encoded = events[index].entries[key]; return encoded === null ? undefined : JSON.parse(encoded);
    }
  }
  return undefined;
}
