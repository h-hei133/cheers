function members(root) {
  const members = [], pending = [root];
  while (pending.length) {
    const dir = pending.pop();
    for (const entry of fs.readdirSync(dir, { withFileTypes: true })) {
      const file = path.join(dir, entry.name);
      if (entry.isDirectory()) pending.push(file);
      else if (entry.isFile()) members.push([path.relative(root, file), sha256(fs.readFileSync(file))]);
    }
  }
  return members.sort(([a], [b]) => a.localeCompare(b));
}
function capture(root) {
  return sha256(JSON.stringify(members(root)));
}
