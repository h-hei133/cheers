import fs from 'node:fs';
import path from 'node:path';
import assert from 'node:assert/strict';
import { spawnSync } from 'node:child_process';
import { createHash } from 'node:crypto';
import { fileURLToPath } from 'node:url';
const directory = path.dirname(fileURLToPath(import.meta.url)), repo = path.resolve(directory, '../..');
const packageRoot = path.join(repo, 'packages/ts-shell');
const hash = value => createHash('sha256').update(value).digest('hex');
const cases = [
  { name: 'route-acceptance-order', file: 'turn/step.ts',
    before: '        prepared.acceptRoute?.();', after: '        /* reversed route acceptance */',
    tests: ['src/cli/provider/route.test.ts'], filter: 'persists accepted route state before compaction' },
  { name: 'route-read-cas', file: 'session/model/routing.ts',
    before: '    store.commit({ [key]: captured === undefined ? initial ?? null : JSON.stringify(captured) }, admit);',
    after: '    if (captured !== undefined) store.commit({ [key]: JSON.stringify(captured) }, admit);',
    tests: ['src/cli/provider/route.test.ts'], filter: 'router state race' },
  { name: 'direct-stateless', file: 'turn/context/route.ts',
    before: "const state = reason === 'direct' ? undefined : request.captureRouteState?.(provider, modelId, admit);",
    after: 'const state = request.captureRouteState?.(provider, modelId, admit);',
    tests: ['src/cli/runtime/compaction/virtual.test.ts'], filter: 'routes (manual|branch) summaries once' },
  { name: 'delegated-selection', file: 'cli/provider/step.ts',
    before: '      virtualModel, virtualModels: model ? undefined : request.virtualModels,',
    after: '      virtualModel: request.virtualModel, virtualModels: request.virtualModels,',
    tests: ['src/cli/provider/route.test.ts'], filter: 'explicit delegated' },
  { name: 'retry-context', file: 'turn/step.ts',
    before: '      failedRoute: () => failedRoute, replayBinding });',
    after: '      failedRoute: () => options.failedRoute, replayBinding });',
    tests: ['src/turn/request.test.ts'], filter: 'implicit-effort retry' },
  { name: 'retry-dispatch-identity', file: 'turn/step.ts',
    before: '      dispatchedOrigin = undefined;', after: '      /* reversed dispatch identity reset */',
    tests: ['src/turn/request.test.ts'], filter: 'previous dispatch marker' },
  { name: 'summary-effort', file: 'cli/runtime/compaction/direct.ts',
    before: "  const effort = routing.route?.thinkingLevel ?? rawEffort;",
    after: "  const effort = routing.route?.thinkingLevel;",
    tests: ['src/cli/runtime/compaction/virtual.test.ts'], filter: 'preserves selected effort.*branch' },
  { name: 'summary-output', file: 'cli/runtime/compaction/direct.ts',
    before: '  if (failure) throw new Error(failure);', after: '  void failure;',
    tests: ['src/cli/runtime/compaction/virtual.test.ts'], filter: 'output budget cannot carry' },
  { name: 'selection-capture', file: 'core/commands/model/virtual.ts',
    before: 'decodeVirtualModel(snapshotData(target, { bytes: 4096, nodes: 8, depth: 2 }))',
    after: 'decodeVirtualModel(target)',
    tests: ['src/core/commands/model/virtual.test.ts'], filter: 'caller-owned effort accessors' },
  { name: 'ui-settlement', file: 'extensions/runtime/virtualSource.ts',
    before: "    try { entry.signal?.removeEventListener('abort', entry.cancel); } catch {}",
    after: "    entry.signal?.removeEventListener('abort', entry.cancel);",
    tests: ['src/extensions/runtime/virtual.test.ts'], filter: 'settles a file UI reply' },
];
const wanted = new Set(process.argv.slice(2)), receipts = [];
for (const item of cases.filter(item => wanted.size === 0 || wanted.has(item.name))) {
  const target = path.join(packageRoot, 'src', item.file), original = fs.readFileSync(target), source = original.toString();
  assert.equal(source.split(item.before).length - 1, 1, 'Unique reversal seam required: ' + item.name);
  let result;
  try {
    fs.writeFileSync(target, source.replace(item.before, item.after));
    result = spawnSync('rtk', ['env', 'VITEST_MAX_THREADS=2', 'VITEST_MIN_THREADS=2', '../../node_modules/.bin/vitest',
      'run', ...item.tests, '--maxWorkers=1', '--no-file-parallelism', '-t', item.filter], {
      cwd: packageRoot, encoding: 'utf8', timeout: 55_000, maxBuffer: 1024 * 1024,
    });
  } finally { fs.writeFileSync(target, original); }
  const log = (result.stdout ?? '') + (result.stderr ?? ''); fs.writeFileSync(path.join(directory, 'reversal-' + item.name + '.log'), log);
  const receipt = { name: item.name, file: item.file, sha256: hash(original), exitCode: result.status, signal: result.signal,
    assertionFailure: result.status === 1 && /AssertionError/.test(log), restoredExact: fs.readFileSync(target).equals(original) };
  receipts.push(receipt); fs.writeFileSync(path.join(directory, 'reversals.json'), JSON.stringify(receipts, null, 2) + '\n');
  console.log(JSON.stringify(receipt)); assert.ok(receipt.assertionFailure); assert.ok(receipt.restoredExact);
}
