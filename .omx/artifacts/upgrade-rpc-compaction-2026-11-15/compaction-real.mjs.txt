import fs from 'node:fs';
import path from 'node:path';
import os from 'node:os';
import assert from 'node:assert/strict';
import { spawn } from 'node:child_process';
import { createHash } from 'node:crypto';
import { fileURLToPath, pathToFileURL } from 'node:url';
import { parseTree, findNodeAtLocation } from 'jsonc-parser';
const directory = path.dirname(fileURLToPath(import.meta.url)), repo = path.resolve(directory, '../..');
const { getRuntimeConfig } = await import(pathToFileURL(path.join(repo, 'packages/ts-shell/dist/index.js')).href);
// Capture configured credentials before creating the isolated runtime home.
const configured = getRuntimeConfig(repo);
const catalogs = new Map();
for (const name of ['models.json', 'models-cache.json']) {
  const file = path.join(os.homedir(), '.cheers', name);
  if (fs.existsSync(file)) catalogs.set(name, fs.readFileSync(file));
}
// Recover only complete JSON provider objects from the malformed user file.
// Choose an already connected configured record; do not invent model metadata.
const rawCatalog = catalogs.get('models.json')?.toString('utf8') ?? '';
const providerNodes = findNodeAtLocation(parseTree(rawCatalog, [], { allowTrailingComma: true }), ['providers']);
const validProviders = [];
for (const property of providerNodes?.children ?? []) {
  const id = property.children?.[0]?.value, value = property.children?.[1];
  try { validProviders.push({ id, ...JSON.parse(rawCatalog.slice(value.offset, value.offset + value.length)) }); } catch {}
}
const originalActive = configured.providerConnections.find(connection => connection.id === configured.activeProviderConnectionId);
let minimalConfiguredModelEntry = false;
for (const property of providerNodes?.children ?? []) {
  if (!originalActive) break;
  const id = property.children?.[0]?.value, node = property.children?.[1];
  if (validProviders.some(provider => provider.id === id)) continue;
  const fields = {};
  for (const entry of node?.children ?? []) {
    const key = entry.children?.[0]?.value, value = entry.children?.[1];
    if (!['label', 'runtime', 'endpoint', 'auth', 'baseUrl', 'envVar'].includes(key)) continue;
    try { fields[key] = JSON.parse(rawCatalog.slice(value.offset, value.offset + value.length)); } catch {}
  }
  if (fields.runtime === originalActive.provider && fields.baseUrl?.replace(/\/+$/, '') === originalActive.baseUrl?.replace(/\/+$/, '')
    && fields.endpoint && fields.auth) {
    // The active model list is malformed. The isolated fixture declares only
    // its already configured identity; it asserts no new capability metadata.
    validProviders.push({ id, ...fields, models: [{ id: originalActive.model }] });
    minimalConfiguredModelEntry = true;
  }
}
const connections = [...configured.providerConnections].sort((a, b) => Number(b.id === configured.activeProviderConnectionId) - Number(a.id === configured.activeProviderConnectionId));
const candidates = connections.filter(connection => connection.status === 'connected' && validProviders.some(provider =>
  (provider.runtime ?? provider.id) === connection.provider
  && provider.baseUrl?.replace(/\/+$/, '') === connection.baseUrl?.replace(/\/+$/, '')
  && provider.models.some(model => model.id === connection.model)));
const candidateIndex = 0;
const selectedConnection = candidates[candidateIndex];
const receipt = { platform: process.platform, compiledCliRpc: true, realProvider: true, status: 'not_run',
  buildId: JSON.parse(fs.readFileSync(path.join(repo, 'packages/ts-shell/dist/build-manifest.json'))).buildId,
  probeSha256: createHash('sha256').update(fs.readFileSync(fileURLToPath(import.meta.url))).digest('hex'), stages: {} };
const root = fs.mkdtempSync(path.join(directory, 'real-state-')), home = path.join(root, 'home'), workspace = path.join(root, 'work');
fs.mkdirSync(home, { mode: 0o700 }); fs.mkdirSync(workspace);
const storage = path.join(home, '.cheers'); fs.mkdirSync(storage, { mode: 0o700 });
const observed = path.join(root, 'requests.jsonl');
const include = 'SHELL_CONTEXT_742619', exclude = 'SHELL_PRIVATE_983267';
let stage = 'setup', child, closed, timer, sequence = 0, lines = '', invalid = 0;
const frames = [], waiters = new Set();
const delay = ms => new Promise(resolve => setTimeout(resolve, ms));
function readFacts() { return fs.existsSync(observed) ? fs.readFileSync(observed, 'utf8').trim().split('\n').filter(Boolean).map(JSON.parse) : []; }
function waitFor(predicate, from = frames.length) {
  const found = frames.slice(from).find(predicate); if (found) return Promise.resolve(found);
  return new Promise((resolve, reject) => {
    const deadline = setTimeout(() => { waiters.delete(waiter); reject(new Error('Response deadline.')); }, 75_000);
    const waiter = { predicate, resolve: value => { clearTimeout(deadline); waiters.delete(waiter); resolve(value); } }; waiters.add(waiter);
  });
}
function command(type, values = {}) {
  const id = String(++sequence), result = waitFor(value => value.type === 'response' && value.id === id);
  child.stdin.write(JSON.stringify({ id, type, ...values }) + '\n'); return result;
}
try {
  assert.equal(process.platform, 'linux'); assert.ok(selectedConnection, 'Configured model selection unavailable.');
  fs.writeFileSync(path.join(storage, 'auth.json'), JSON.stringify({ provider: configured.provider,
    providerCredentials: configured.providerCredentials, providerConnections: configured.providerConnections,
    activeProviderConnectionId: selectedConnection.id, accountPool: configured.accountPool }), { mode: 0o600 });
  fs.writeFileSync(path.join(storage, 'models.json'), JSON.stringify({ providers: Object.fromEntries(validProviders.map(provider => [provider.id, provider])) }), { mode: 0o600 });
  fs.writeFileSync(path.join(storage, 'config.json'), JSON.stringify({ agentMode: 'build', maxOutputTokens: 1024,
    reserveTokens: 0, keepRecentTokens: 80, compactionEnabled: false, retryEnabled: false, providerRetryMaxRetries: 0, cacheWarming: 'off',
    providerConnectTimeoutMs: 10_000, providerHeadersTimeoutMs: 60_000, providerBodyTimeoutMs: 60_000 }));
  receipt.catalogueSetup = { originalActiveSelected: selectedConnection.id === configured.activeProviderConnectionId,
    existingConfiguredTarget: true, originalConfigUnchanged: true, minimalConfiguredModelEntry, isolatedValidObjects: true };
  const observer = path.join(root, 'observer.mjs');
  fs.writeFileSync(observer, `import fs from 'node:fs'; import { channel } from 'node:diagnostics_channel';\nconst write = value => fs.appendFileSync(${JSON.stringify(observed)}, JSON.stringify(value)+'\\n');\nchannel('undici:request:bodySent').subscribe(({request}) => { if(request.method === 'POST') write({bodySent:true}); });\nchannel('undici:request:create').subscribe(({request}) => { if(request.method !== 'POST') return; const body = typeof request.body === 'string' ? request.body : Buffer.isBuffer(request.body) ? request.body.toString('utf8') : undefined; write({created:true,bodyInspectable:body!==undefined,included:body?.includes(${JSON.stringify(include)}),excluded:body?.includes(${JSON.stringify(exclude)})}); });\n`);
  const env = { ...process.env, HOME: home, USERPROFILE: home, CHEERS_TEST_ROOT: home, GIT_CEILING_DIRECTORIES: root,
    CHEERS_PROVIDER_MODE: 'real', CHEERS_REAL_PROVIDER_SMOKE: '1',
    CHEERS_MARKETPLACES_FILE: path.join(home, 'sources.json'), CHEERS_EXTENSIONS_FILE: path.join(home, 'extensions.json'),
    NODE_OPTIONS: '--import ' + pathToFileURL(observer).href };
  child = spawn('rtk', ['node', path.join(repo, 'scripts/start.mjs'), '--rpc', '--project', workspace,
    '--no-context-files', '--no-skills', '--tools', 'bash,read', '--non-interactive-approval', 'allow-once'],
    { cwd: workspace, env, stdio: ['pipe', 'pipe', 'pipe'] });
  child.stderr.on('data', () => {});
  closed = new Promise(resolve => child.on('close', (code, signal) => resolve({ code, signal })));
  child.stdout.on('data', bytes => {
    lines += bytes.toString();
    for (;;) {
      const newline = lines.indexOf('\n'); if (newline < 0) break;
      const line = lines.slice(0, newline); lines = lines.slice(newline + 1); if (!line.trim()) continue;
      let frame; try { frame = JSON.parse(line); } catch { invalid++; continue; }
      frames.push(frame); for (const waiter of [...waiters]) if (waiter.predicate(frame)) waiter.resolve(frame);
    }
  });
  timer = setTimeout(() => child.kill('SIGTERM'), 160_000);
  const initial = await command('get_state'); assert.equal(initial.success, true);
  stage = 'seed'; const sessionId = initial.data.sessionId;
  assert.equal((await command('set_thinking_level', { level: 'auto' })).success, true);
  const visible = await command('bash', { command: "printf 'RPC_OLDER_CONTEXT'; printf '%08192d' 0" });
  assert.equal(visible.success, true); assert.equal(visible.data.historySaved, 'saved');
  assert.equal((await command('bash', { command: 'printf RPC_ADDITIONAL_CONTEXT' })).success, true);
  const ended = waitFor(frame => frame.type === 'agent_end');
  assert.equal((await command('prompt', { message: 'Return exactly RPC_COMPACTION_READY and no tool calls. ' + 'This is a bounded compaction verification sample. '.repeat(80) })).success, true);
  const outcome = await ended; receipt.promptStatus = outcome.status; assert.equal(outcome.status, 'completed');
  const before = readFacts().filter(fact => fact.bodySent).length; assert.ok(before > 0);
  stage = 'compact';
  const compact = command('compact', { customInstructions: 'Preserve the single test fact briefly. Return one short summary.' });
  const state = await command('get_state'); assert.equal(state.success, true);
  const result = await compact; assert.equal(result.success, true); assert.equal(result.data.status, 'compacted');
  assert.ok(result.data.tokensAfter < result.data.tokensBefore);
  const events = fs.readFileSync(path.join(storage, 'sessions', sessionId, 'transcript.jsonl'), 'utf8').trim().split('\n').map(JSON.parse);
  assert.equal(events.filter(event => event.type === 'compaction-commit').length, 1);
  const facts = readFacts(); receipt.requests = facts.filter(fact => fact.bodySent).length;
  assert.ok(receipt.requests > before);
  receipt.wireObservation = facts.filter(fact => fact.created).map(fact => ({ created: true, bodyInspectable: fact.bodyInspectable }));
  receipt.stages.compact = { passed: true, actualSummaryBodySent: true, durableCompaction: true,
    stateControlResponded: true, structuredOutcome: result.data.status };
  stage = 'eof'; child.stdin.end(); assert.deepEqual(await closed, { code: 0, signal: null }); assert.equal(invalid, 0);
  receipt.stages.eof = { passed: true, exit0: true, validJsonLines: true }; receipt.status = 'passed';
} catch(error) {
  receipt.status = 'failed'; receipt.failedStage = stage; receipt.requests = readFacts().filter(fact => fact.bodySent).length;
  receipt.failure = { name: error?.name, code: error?.code, message: 'Configured CLI probe failed; provider details withheld.' };
  process.exitCode = 1;
} finally {
  clearTimeout(timer); if(child && child.exitCode === null) { child.kill('SIGTERM'); await Promise.race([closed,delay(5000)]); }
  fs.writeFileSync(path.join(directory, 'compaction-real.json'), JSON.stringify(receipt,null,2)+'\n'); console.log(JSON.stringify(receipt));
  fs.rmSync(root,{recursive:true,force:true});
}
