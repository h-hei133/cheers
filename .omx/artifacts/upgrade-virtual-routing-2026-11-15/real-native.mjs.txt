import fs from 'node:fs';
import path from 'node:path';
import os from 'node:os';
import assert from 'node:assert/strict';
import { createHash } from 'node:crypto';
import { channel } from 'node:diagnostics_channel';
import { fileURLToPath, pathToFileURL } from 'node:url';
import { parseTree, findNodeAtLocation } from 'jsonc-parser';

const directory = path.dirname(fileURLToPath(import.meta.url));
const repo = path.resolve(directory, '../..');
const installed = path.join(directory, 'install/node_modules/@cheers/ts-shell');
const { createAgentSession, DefaultResourceLoader, getRuntimeConfig } = await import(pathToFileURL(path.join(installed, 'dist/index.js')).href);
const { catalogProvider } = await import(pathToFileURL(path.join(installed, 'dist/core/models/catalog.js')).href);
// Capture configured credentials before creating the isolated runtime home.
const configured = getRuntimeConfig(repo);
const catalogs = new Map();
for (const name of ['models.json', 'models-cache.json']) {
  const file = path.join(os.homedir(), '.cheers', name);
  if (fs.existsSync(file)) catalogs.set(name, fs.readFileSync(file));
}
// Recover only complete JSON provider objects from the malformed user file.
// Choose an already connected configured record; do not invent model metadata.
const rawCatalog = catalogs.get('models.json')?.toString('utf8') ?? '';
const providerNodes = findNodeAtLocation(parseTree(rawCatalog, [], { allowTrailingComma: true }), ['providers']);
const validProviders = [];
for (const property of providerNodes?.children ?? []) {
  const id = property.children?.[0]?.value, value = property.children?.[1];
  try { validProviders.push({ id, ...JSON.parse(rawCatalog.slice(value.offset, value.offset + value.length)) }); } catch {}
}
const originalActive = configured.providerConnections.find(connection => connection.id === configured.activeProviderConnectionId);
let minimalConfiguredModelEntry = false;
for (const property of providerNodes?.children ?? []) {
  if (!originalActive) break;
  const id = property.children?.[0]?.value, node = property.children?.[1];
  if (validProviders.some(provider => provider.id === id)) continue;
  const fields = {};
  for (const entry of node?.children ?? []) {
    const key = entry.children?.[0]?.value, value = entry.children?.[1];
    if (!['label', 'runtime', 'endpoint', 'auth', 'baseUrl', 'envVar'].includes(key)) continue;
    try { fields[key] = JSON.parse(rawCatalog.slice(value.offset, value.offset + value.length)); } catch {}
  }
  if (fields.runtime === originalActive.provider && fields.baseUrl?.replace(/\/+$/, '') === originalActive.baseUrl?.replace(/\/+$/, '')
    && fields.endpoint && fields.auth) {
    // The active model list is malformed. The isolated fixture declares only
    // its already configured identity; it asserts no new capability metadata.
    validProviders.push({ id, ...fields, models: [{ id: originalActive.model }] });
    minimalConfiguredModelEntry = true;
  }
}
const connections = [...configured.providerConnections].sort((a, b) => Number(b.id === configured.activeProviderConnectionId) - Number(a.id === configured.activeProviderConnectionId));
const candidates = connections.filter(connection => connection.status === 'connected' && validProviders.some(provider =>
  (provider.runtime ?? provider.id) === connection.provider
  && provider.baseUrl?.replace(/\/+$/, '') === connection.baseUrl?.replace(/\/+$/, '')
  && provider.models.some(model => model.id === connection.model)));
const candidateIndex = Number(process.env.CHEERS_REAL_TARGET_INDEX ?? '0');
const selectedConnection = candidates[candidateIndex];
const manifest = JSON.parse(fs.readFileSync(path.join(installed, 'dist/build-manifest.json')));
const receipt = { platform: process.platform, node: process.version, status: 'not_run', publicInstalledSdk: true,
  configuredProvider: true, buildId: manifest.buildId, requests: 0, stages: {},
  probeSha256: createHash('sha256').update(fs.readFileSync(fileURLToPath(import.meta.url))).digest('hex') };
const previous = new Map(), sessions = new Set(), loaders = new Set();
const root = fs.mkdtempSync(path.join(directory, 'real-state-'));
const home = path.join(root, 'home'), workspace = path.join(root, 'work');
fs.mkdirSync(home, { mode: 0o700 }); fs.mkdirSync(workspace);
const storage = path.join(home, '.cheers'); fs.mkdirSync(storage, { mode: 0o700 });
let stage = 'setup', session, timer, lastFailure;
let target;
const seen = [];
const sent = channel('undici:request:bodySent');
const onSent = ({ request }) => {
  if (request.method !== 'POST') return;
  receipt.requests++;
  receipt.stages[stage] = { ...receipt.stages[stage], posts: (receipt.stages[stage]?.posts ?? 0) + 1 };
  console.log(JSON.stringify({ stage, bodySent: true, request: receipt.requests }));
};
function events(current) {
  return current.sessionManager.getTranscriptEvents(current.sessionId).filter(event => event.type === 'branch-store' && event.namespace === 'model-routing');
}
function state(current) {
  const key = JSON.stringify(['adaptive', 'configured']);
  const rows = events(current);
  for (let index = rows.length - 1; index >= 0; index--) {
    if (Object.hasOwn(rows[index].entries, key)) return JSON.parse(rows[index].entries[key]);
  }
}
try {
  assert.equal(process.platform, 'linux');
  assert.ok(selectedConnection, 'A configured connected target needs a complete existing catalogue object.');
  assert.equal(manifest.buildId, JSON.parse(fs.readFileSync(path.join(repo, 'packages/ts-shell/dist/build-manifest.json'))).buildId);
  for (const [key, value] of Object.entries({ HOME: home, USERPROFILE: home, CHEERS_TEST_ROOT: home,
    GIT_CEILING_DIRECTORIES: root, CHEERS_PROVIDER_MODE: 'real', CHEERS_REAL_PROVIDER_SMOKE: '1',
    CHEERS_MARKETPLACES_FILE: path.join(home, 'sources.json'), CHEERS_EXTENSIONS_FILE: path.join(home, 'extensions.json') })) {
    previous.set(key, process.env[key]); process.env[key] = value;
  }
  fs.writeFileSync(path.join(storage, 'auth.json'), JSON.stringify({ provider: configured.provider,
    providerCredentials: configured.providerCredentials, providerConnections: configured.providerConnections,
    activeProviderConnectionId: selectedConnection.id, accountPool: configured.accountPool }), { mode: 0o600 });
  for (const [name, contents] of catalogs) fs.writeFileSync(path.join(storage, name), contents, { mode: 0o600 });
  fs.writeFileSync(path.join(storage, 'models.json'), JSON.stringify({ providers: Object.fromEntries(validProviders.map(provider => [provider.id, provider])) }), { mode: 0o600 });
  receipt.catalogueSetup = { isolatedValidObjects: true, originalActiveSelected: selectedConnection.id === configured.activeProviderConnectionId,
    existingConfiguredTarget: true, originalConfigUnchanged: true, candidateIndex, minimalConfiguredModelEntry };
  const config = { ...getRuntimeConfig(workspace), contextWindow: 32_000, keepRecentTokens: 400, reserveTokens: 0,
    maxOutputTokens: 1024, compactionEnabled: false, cacheWarmIntervalSeconds: 0, retryEnabled: false,
    retryMaxRetries: 0, providerRetryMaxRetries: 0,
    providerConnectTimeoutMs: 10_000, providerHeadersTimeoutMs: 45_000, providerBodyTimeoutMs: 45_000 };
  const marker = 'ROUTING_SESSION_ACK_63852';
  const makeLoader = () => {
    const loader = new DefaultResourceLoader({ cwd: workspace, noContextFiles: true, noSkills: true, noThemes: true,
      systemPrompt: `Return only ${marker}. When summarizing, preserve that marker and use fewer than 60 words.`,
      extensionFactories: [api => api.registerVirtualModel({ provider: 'adaptive', id: 'configured', name: 'Configured', route: request => {
        assert.ok(target);
        seen.push({ reason: request.reason, count: request.state?.count, previousMatches: request.previous?.modelId === target.modelId });
        return { ...target, state: { count: (request.state?.count ?? 0) + 1 } };
      } })] });
    loaders.add(loader); return loader;
  };
  const create = async extra => {
    const loader = makeLoader(); await loader.reload(); assert.equal(loader.getExtensions().errors.length, 0);
    const result = await createAgentSession({ cwd: workspace, runtimeConfig: config, resourceLoader: loader, noTools: 'all', ...extra });
    sessions.add(result.session);
    result.session.subscribe(event => { if (event.type === 'query.failed') lastFailure = String(event.error); });
    return result.session;
  };
  session = await create({ model: 'adaptive/configured' });
  const resolution = session.runtimeContext.resolveConfigTarget(session.runtimeContext.runtimeConfig, session.sessionId);
  assert.equal(resolution.blocked, undefined); assert.ok(resolution.credential); assert.ok(resolution.model);
  const owner = catalogProvider(session.getModelCatalog().physical, resolution.provider, resolution.baseUrl ?? resolution.credential.baseUrl);
  assert.equal(owner.ambiguous, false); assert.ok(owner.entry?.models.some(model => model.id === resolution.model));
  target = { provider: owner.entry.id, modelId: resolution.model };
  receipt.configuredPhysicalTargetResolved = true;
  sent.subscribe(onSent);
  timer = setTimeout(() => { receipt.deadline = true; void session?.abort().catch(() => {}); }, 190_000);
  stage = 'prompt';
  assert.equal((await session.prompt(`Return only ${marker}.`)).trim(), marker);
  assert.equal(state(session)?.count, 1);
  assert.ok(session.getState().turns.some(message => message.role === 'assistant' && message.origin?.provider === resolution.provider && message.origin?.model === target.modelId));
  receipt.stages.prompt = { ...receipt.stages.prompt, passed: true, physicalOrigin: true, stateAccepted: true };
  stage = 'manual-summary';
  for (let index = 0; index < 5; index++) {
    session.sessionManager.persistUserPromptBarrier(session.sessionId, `Recorded work ${index}: ${'verified bounded facts '.repeat(100)}`);
    session.sessionManager.addTurn(session.sessionId, { role: 'assistant', content: 'Saved decision '.repeat(60) });
  }
  const before = events(session).length;
  assert.equal((await session.compact(`Keep ${marker}; use fewer than 60 words.`)).status, 'compacted');
  assert.equal(seen.at(-1).reason, 'direct'); assert.equal(seen.at(-1).count, undefined);
  assert.equal(events(session).length, before); assert.equal(state(session)?.count, 1);
  receipt.stages[stage] = { ...receipt.stages[stage], passed: true, directStateless: true };
  stage = 'branch-summary';
  const summary = await session.sessionRuntime.summarizeAbandonedTurns([
    { role: 'user', content: `${marker}: ${'abandoned verified decisions '.repeat(70)}` },
    { role: 'assistant', content: 'Retain the result '.repeat(60) },
  ], new AbortController().signal);
  assert.ok(summary.includes('<compacted-summary>')); assert.ok(summary.includes(marker));
  assert.equal(seen.at(-1).reason, 'direct'); assert.equal(seen.at(-1).count, undefined);
  assert.equal(events(session).length, before); assert.equal(state(session)?.count, 1);
  receipt.stages[stage] = { ...receipt.stages[stage], passed: true, directStateless: true, surface: 'session.sessionRuntime' };
  stage = 'reopen';
  const id = session.sessionId; await session.dispose(); sessions.delete(session);
  session = await create({ sessionId: id });
  assert.equal((await session.prompt(`Return only ${marker}.`)).trim(), marker);
  assert.equal(seen.at(-1).count, 1); assert.equal(state(session)?.count, 2);
  receipt.stages[stage] = { ...receipt.stages[stage], passed: true, stateReused: true };
  assert.equal(receipt.requests, 4);
  receipt.status = 'passed';
} catch (error) {
  receipt.status = 'failed'; receipt.failedStage = stage; receipt.errorName = error?.name;
  receipt.line = String(error?.stack).match(/real-native\.mjs:(\d+)/)?.[1];
  const detail = `${String(error?.message ?? '')} ${lastFailure ?? ''}`;
  receipt.failure = /quota|429/i.test(detail) ? 'provider-quota' : /auth|credential|401|403/i.test(detail) ? 'provider-auth'
    : /network|fetch|timeout|ECONN|EHOST|ENOTFOUND/i.test(detail) ? 'provider-network' : /catalog|target|connection/i.test(detail) ? 'target-admission' : 'product-or-contract';
  receipt.failureFacts = Object.fromEntries(Object.entries({
    connectionUnavailable: /connection is unavailable/i, noBinding: /no usable provider binding/i,
    noCredential: /no usable credential/i, endpointOwnership: /catalogue does not own/i,
    configuredLimits: /configured request limits/i, authorityChanged: /authority changed|changed during|retired/i,
    configurationChanged: /configuration|settings.*changed/i, sourceUnavailable: /unavailable/i,
    fetchFailed: /fetch failed/i, refused: /ECONNREFUSED|connection refused/i, timeout: /timeout|timed out/i,
    thinking: /thinking|reasoning/i, history: /history|branch/i, modelAdmission: /model.*admission/i,
  }).map(([key, value]) => [key, value.test(detail)]));
  receipt.errorCodes = [];
  for (let item = error, depth = 0; item && depth < 4; item = item.cause, depth++) {
    if (typeof item.code === 'string' && /^[A-Z][A-Z_0-9]{1,50}$/.test(item.code)) receipt.errorCodes.push(item.code);
  }
  if (error?.name === 'AssertionError') receipt.assertion = { operator: error.operator,
    actual: typeof error.actual === 'number' || typeof error.actual === 'boolean' ? error.actual : undefined,
    expected: typeof error.expected === 'number' || typeof error.expected === 'boolean' ? error.expected : undefined };
  process.exitCode = 1;
} finally {
  clearTimeout(timer);
  for (const current of sessions) await current.dispose().catch(() => {});
  for (const loader of loaders) { try { loader.captureRetirement()(); } catch {} }
  sent.unsubscribe(onSent);
  for (const [key, value] of previous) { if (value === undefined) delete process.env[key]; else process.env[key] = value; }
  fs.rmSync(root, { recursive: true, force: true });
  receipt.isolatedStateRemoved = !fs.existsSync(root);
  fs.writeFileSync(path.join(directory, 'real-native.json'), JSON.stringify(receipt, null, 2) + '\n');
  console.log(JSON.stringify(receipt));
}
