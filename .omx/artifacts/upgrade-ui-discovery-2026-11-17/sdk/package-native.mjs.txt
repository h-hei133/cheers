async function run(code, command, args, cwd, timeout = 120_000) {
  stage = code;
  const child = spawn(command, args, { cwd, env: { ...process.env }, stdio: ['ignore', 'pipe', 'pipe'] }); children.add(child);
  let stdout = '', stderr = '', bytes = 0, overflow = false, timedOut = false, hardKill;
  const collect = (chunk, output) => {
    bytes += chunk.length;
    if (bytes > 8 * 1024 * 1024) { overflow = true; child.kill('SIGKILL'); return; }
    if (output) stdout += chunk.toString('utf8'); else stderr += chunk.toString('utf8');
  };
  child.stdout.on('data', chunk => collect(chunk, true)); child.stderr.on('data', chunk => collect(chunk, false));
  const timer = setTimeout(() => { timedOut = true; child.kill('SIGTERM'); hardKill = setTimeout(() => child.kill('SIGKILL'), 750); }, timeout);
  try {
    const result = await new Promise(resolve => {
      child.once('error', () => resolve({ exit: null, signal: 'spawn_error' }));
      child.once('close', (exit, signal) => resolve({ exit, signal }));
    });
    receipt.commands.push({ code, ...result, outputBytes: bytes, stdoutSha256: hash(stdout), stderrSha256: hash(stderr), timedOut, overflow });
    check(result.exit === 0 && !result.signal && !timedOut && !overflow, `${code}-command-failed`);
    return stdout;
  } finally { clearTimeout(timer); clearTimeout(hardKill); children.delete(child); }
}

    type: 'module', dependencies: { [packageJson.name]: 'file:' + archive, typescript: exact('typescript'),
      '@types/node': exact('@types/node'), '@types/picomatch': exact('@types/picomatch'), '@types/which': exact('@types/which') } }), { mode: 0o600 });
  await run('install', 'npm', ['install', '--ignore-scripts', '--no-audit', '--no-fund', '--prefer-offline'], install, 240_000);
  installedRoot = path.join(install, 'node_modules/@cheers/ts-shell');
  check(!fs.lstatSync(installedRoot).isSymbolicLink(), 'installed-package-copy');
  check(hash(fs.readFileSync(path.join(installedRoot, 'dist/build-manifest.json'))) === receipt.buildManifestSha256, 'installed-current-manifest');

  const imports = entries.map((name, index) => 'import * as entry' + index + ' from ' + JSON.stringify(name) + ';\nvoid entry' + index + ';').join('\n');

    files: ['./consumer.mts', './ui-file.ts'] }));

    resourceBinding: true, nativeSignalPositive: true, observationSignalNegative: true, invalidInputReturnNegative: true, publicExtensionService: true };

} catch (error) { receipt.status = 'failed'; receipt.failure = { stage, code: error?.probeCode ?? 'unexpected-error' }; process.exitCode = 1; }
finally {
  for (const child of children) child.kill('SIGKILL');
  for (const [key, value] of previousEnv) { if (value === undefined) delete process.env[key]; else process.env[key] = value; }
  try {
    receipt.originalInputsChecked = originals.size;
    receipt.originalInputsUnchanged = [...originals].every(([file, before]) => (fs.existsSync(file) ? hash(fs.readFileSync(file)) : null) === before);
    receipt.sourceUnchanged = beforeSources !== undefined && canonical(sources()) === canonical(beforeSources);
    receipt.compiledPackageUnchanged = beforeManifest !== undefined && fs.readFileSync(path.join(packageRoot, 'dist/build-manifest.json')).equals(beforeManifest);

  } catch { receipt.privateStateRemoved = false; }
  if (!receipt.originalInputsUnchanged || !receipt.sourceUnchanged || !receipt.compiledPackageUnchanged || !receipt.privateStateRemoved) {
    receipt.status = 'failed'; receipt.cleanupFailure = 'private-state-or-original-identity'; process.exitCode = 1;
  }
  if (receiptFile.startsWith(path.join(repo, '.tmp') + path.sep)) {
    fs.mkdirSync(path.dirname(receiptFile), { recursive: true }); fs.writeFileSync(receiptFile, JSON.stringify(receipt, null, 2) + '\n', { mode: 0o600 });
  }
  process.stdout.write(JSON.stringify(receipt, null, 2) + '\n');
}
