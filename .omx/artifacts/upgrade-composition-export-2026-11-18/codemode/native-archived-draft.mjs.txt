async function bound(operation, code, ms = SCENARIO_MS, retire) {
  let timer;
  const promise = Promise.resolve(operation);
  void promise.catch(() => {});
  try {
    return await Promise.race([promise, new Promise((_, reject) => {
      timer = setTimeout(() => {
        if (retire) void Promise.resolve(retire()).catch(() => {});
        const error = new Error(code); error.probeCode = code; reject(error);
      }, ms);
    })]);
  } finally { clearTimeout(timer); }
}

  return { promise, resolve };
}
async function until(predicate, code, ms = 3000) {

    await new Promise(resolve => setTimeout(resolve, 10));
  }
}

function nativeSummary(output) {

  for (const key of ['collisionSafe', 'namespace', 'physical', 'logical', 'missing', 'frozen', 'callbacks'])

}
function wireProjection(body) {
  const declarations = [], results = [], system = [], schemas = [];
  const addDeclaration = (name, parameters) => {
    if (typeof name !== 'string' || !parameters || typeof parameters !== 'object') return;
    const json = JSON.stringify(parameters);
    declarations.push({ name, schemaSha256: hash(json), schemaBytes: Buffer.byteLength(json) });
    schemas.push({ name, parameters });
  };
  const addResult = (id, content) => {
    const value = resultFact(id, content);
    try {
      const parsed = JSON.parse(textOf(content));
      if (parsed?.query === SCRIPT && Array.isArray(parsed.loaded)) value.search = {
        queryExact: true, loadedScript: parsed.loaded.includes(SCRIPT),
        matches: Array.isArray(parsed.matches) ? parsed.matches.map(match => ({
          name: typeof match?.name === 'string' ? match.name : null,
          loaded: match?.loaded === true, pinned: match?.pinned === true })) : [],
      };
    } catch { /* Only fixture facts and digests are retained. */ }
    results.push(value);
  };
  for (const tool of body.tools ?? []) {
    if (tool.function) addDeclaration(tool.function.name, tool.function.parameters);
    else if (Array.isArray(tool.functionDeclarations)) for (const definition of tool.functionDeclarations)
      addDeclaration(definition.name, definition.parameters ?? definition.parametersJsonSchema);
    else addDeclaration(tool.name, tool.input_schema ?? tool.parameters);
  }
  for (const tool of body.toolConfig?.tools ?? []) addDeclaration(tool.toolSpec?.name, tool.toolSpec?.inputSchema?.json);
  for (const message of body.messages ?? []) {
    if (message.role === 'system') system.push(textOf(message.content));
    if (message.role === 'tool') addResult(message.tool_call_id, message.content);
    for (const block of Array.isArray(message.content) ? message.content : []) {
      if (block.type === 'tool_result') addResult(block.tool_use_id, block.content);
      if (block.toolResult) addResult(block.toolResult.toolUseId, block.toolResult.content);
    }
  }
  for (const item of Array.isArray(body.input) ? body.input : []) {
    if (item.type === 'function_call_output') addResult(item.call_id, item.output);
    if (item.role === 'system') system.push(textOf(item.content));
  }
  for (const content of body.contents ?? []) for (const part of content.parts ?? [])
    if (part.functionResponse) addResult(part.functionResponse.id, part.functionResponse.response?.output);
  if (body.instructions !== undefined) system.push(textOf(body.instructions));
  if (body.system !== undefined) system.push(textOf(body.system));
  if (body.systemInstruction?.parts) system.push(textOf(body.systemInstruction.parts));
  const systemText = system.join('\n');
  return { declarations, toolResults: results, schemaSha256: hash(JSON.stringify(schemas)),
    systemSha256: hash(systemText), systemBytes: Buffer.byteLength(systemText),
    modelMatchesConfigured: typeof body.model === 'string' ? body.model === selected.model : null };
}

function responseFacts(bytes, encoding) {
  const decompress = encoding === 'gzip' ? gunzipSync : encoding === 'deflate' ? inflateSync
    : encoding === 'br' ? brotliDecompressSync : undefined;
  const text = (decompress ? decompress(bytes, { maxOutputLength: MAX_BYTES }) : bytes).toString('utf8');
  let frames = 0, completionMarker = false;
  const capture = frame => {
    if (!frame || typeof frame !== 'object') return;
    frames++;
    completionMarker ||= frame.type === 'response.completed' || frame.type === 'message_stop'
      || typeof frame.stop_reason === 'string' || frame.choices?.some(choice => typeof choice.finish_reason === 'string') === true
      || frame.candidates?.some(candidate => typeof candidate.finishReason === 'string') === true;
  };
  try { const value = JSON.parse(text); for (const frame of Array.isArray(value) ? value : [value]) capture(frame); }
  catch {
    for (const line of text.split(/\r?\n/)) {
      if (!line.startsWith('data:')) continue;
      const value = line.slice(5).trim();
      if (value === '[DONE]') { completionMarker = true; continue; }
      if (!value) continue;
      try { capture(JSON.parse(value)); } catch { /* Malformed provider content is not persisted. */ }
    }
  }
  return { inspectable: frames > 0, frames, completionMarker };
}

function finalize(request, end) {
  const capture = pending.get(request); if (!capture || capture.finalized) return;
  capture.finalized = true; counters.finalized++;
  try {
    check(capture.incomingBytes <= MAX_BYTES, 'response-capture-limit');
    if (capture.observation) Object.assign(capture.observation, {
      responseStatus: capture.statusCode ?? null,
      responseBytes: capture.incomingBytes,
      responseSha256: hash(Buffer.concat(capture.incoming)),
      responseEnd: end,
      response: responseFacts(Buffer.concat(capture.incoming), capture.encoding),
    });
  } catch { observerFailure = 'response-capture'; }
  finally { capture.outgoing.length = 0; capture.incoming.length = 0; captures.delete(capture); pending.delete(request); }
}

function onHeaders({ request, response }) {
  const capture = pending.get(request); if (!capture) return;
  counters.headers++; capture.statusCode = response.statusCode;
  for (let index = 0; index < response.headers.length; index += 2)
    if (String(response.headers[index]).toLowerCase() === 'content-encoding')
      capture.encoding = String(response.headers[index + 1]).toLowerCase();
}

function onCreate({ request }) {
  if (request.method !== 'POST' || String(request.origin) !== targetOrigin) return;
  counters.create++;
  const capture = { stage, outgoing: [], incoming: [], outgoingBytes: 0, incomingBytes: 0 };
  pending.set(request, capture); captures.add(capture);
  const originalBody = request.onBodySent, originalData = request.onData, originalHeaders = request.onHeaders;
  const originalComplete = request.onComplete, originalError = request.onError;
  request.onBodySent = function (chunk) {
    try {
      capture.outgoingBytes += Buffer.byteLength(chunk);
      if (capture.outgoingBytes <= MAX_BYTES) capture.outgoing.push(Buffer.from(chunk));
      else { observerFailure = 'request-capture-limit'; capture.outgoing.length = 0; }
    } catch { observerFailure = 'request-capture'; }
    return originalBody.call(this, chunk);
  };
  request.onData = function (chunk) {
    try {
      counters.data++; capture.incomingBytes += Buffer.byteLength(chunk);
      if (capture.incomingBytes <= MAX_BYTES) capture.incoming.push(Buffer.from(chunk));
      else { observerFailure = 'response-capture-limit'; capture.incoming.length = 0; }
    } catch { observerFailure = 'response-capture'; }
    return originalData.call(this, chunk);
  };
  request.onHeaders = function (statusCode, rawHeaders, ...rest) {
    onHeaders({ request: this, response: { statusCode, headers: rawHeaders } });
    return originalHeaders.call(this, statusCode, rawHeaders, ...rest);
  };
  request.onComplete = function (...args) {
    counters.complete++; finalize(this, 'transport-complete');
    return originalComplete.apply(this, args);
  };
  request.onError = function (...args) {
    counters.retired++;
    // SDK stream cleanup may retire an accepted transport before trailers. Preserve its already captured data.
    // Native query/effect assertions below remain independent; an accepted status is not a success verdict.
    finalize(this, capture.statusCode >= 200 && capture.statusCode < 300 && capture.incomingBytes > 0
      ? 'accepted-stream-retirement' : 'transport-error');
    return originalError.apply(this, args);
  };
}

function onSent({ request }) {
  const capture = pending.get(request); if (!capture) return;
  try {
    check(observations.length < MAX_REQUESTS, 'native-request-limit');
    check(capture.outgoingBytes <= MAX_BYTES, 'request-capture-limit');
    const bytes = Buffer.concat(capture.outgoing), body = JSON.parse(bytes.toString('utf8'));
    counters.bodySent++;
    capture.observation = { stage: capture.stage, bodySent: true, requestBytes: capture.outgoingBytes,
      requestSha256: hash(bytes), ...wireProjection(body) };
    observations.push(capture.observation);
  } catch (error) { observerFailure = error.probeCode ?? 'request-capture'; void session?.abort().catch(() => {}); }
  finally { capture.outgoing.length = 0; }
}

function onTrailers({ request }) { finalize(request, 'transport-complete'); }


function eventObservation(event) {
  if (!caseState) return;
  if (event.type === 'query.completed') {
    caseState.terminalStatus = event.result?.status ?? null;
    const reason = String(event.result?.stopReason ?? event.result?.finalOutput ?? '');
    caseState.terminal = {
      status: event.result?.status ?? null,
      reasonSha256: hash(reason),
      reasonBytes: Buffer.byteLength(reason),
      code: /provider-readiness:network/i.test(reason) ? 'provider-readiness-network'
        : /provider-readiness:auth/i.test(reason) ? 'provider-readiness-auth'
          : /provider-readiness:configuration/i.test(reason) ? 'provider-readiness-configuration'
            : /no.*(?:model|response|reply)|(?:model|response|reply).*no/i.test(reason) ? 'missing-model-response'
        : /limit|budget|context|token/i.test(reason) ? 'request-limits'
          : /admission|declaration|history|changed|stale/i.test(reason) ? 'request-admission'
            : /credential|auth|configured|catalog|provider|model/i.test(reason) ? 'configured-target'
              : /refus|denied|permission|approval/i.test(reason) ? 'tool-refusal' : 'unknown-stop',
    };
    if (event.result?.status !== 'completed') lastFailure = reason;
  }
  if (event.type === 'query.completed_unproven') caseState.unproven = true;
  if (event.type === 'query.failed' || event.type === 'lifecycle.error') lastFailure = String(event.error);
  if (event.type === 'tool.started') caseState.calls.push({ id: event.taskId, name: event.toolName,
    argumentsSha256: hash(JSON.stringify(event.args ?? {})),
    exactScript: event.toolName === SCRIPT ? event.args?.code === caseState.script : null,
    schemaQueryExact: event.toolName === SEARCH ? event.args?.query === SCRIPT : null,
    singleStepLoad: event.toolName === SEARCH ? event.args?.load === true : null });
  if (event.type === 'approval.resolved') caseState.approvals.push({
    id: event.taskId, name: event.toolName, outcome: event.outcome });
  if (event.type === 'tool.result') {
    const text = event.output ?? '';
    const fact = { id: event.taskId, name: event.toolName, success: event.success,
      outputSha256: hash(text), outputBytes: Buffer.byteLength(text),
      selected: text.includes(SELECTED), discarded: text.includes(RAW_A) || text.includes(RAW_B),
      images: event.images?.length ?? 0 };
    if (event.toolName === SCRIPT && event.success) {
      try { fact.summary = nativeSummary(text); }
      catch { caseState.summaryUninspectable = true; }
    }
    if (event.toolName === SEARCH && event.success) {
      try {
        const value = JSON.parse(text);
        fact.search = { queryExact: value.query === SCRIPT, loadedScript: value.loaded?.includes(SCRIPT) === true,
          matches: value.matches?.map(match => ({ name: match.name, loaded: match.loaded === true, pinned: match.pinned === true })) };
      } catch { caseState.searchUninspectable = true; }
    }
    caseState.results.push(fact);
  }
}
function hookObservation(payload, mode) {
  if (!caseState) return { decision: 'allow' };
  const output = payload.output ?? payload.error ?? '';
  caseState.hooks.push({ event: payload.eventName, name: payload.toolName, id: payload.toolCallId,
    argumentsSha256: hash(JSON.stringify(payload.args ?? {})),
    outputSha256: hash(output), discarded: output.includes(RAW_A) || output.includes(RAW_B) });
  if (payload.eventName === 'PreToolUse' && payload.toolName === EFFECT && mode === 'retry') {
    caseState.retryHooks++;
    return { decision: 'retry', reason: RETRY };
  }
  return { decision: 'allow' };
}
function scriptFor(mode) {

    '  written: values[2].includes(fixture.value), inspection };',
  );
  else lines.push(
    'const values = await Promise.allSettled([tools.read({path:"first.json"}), tools.read({path:"second.json"}),',
    '  tools[fixture.effect]({value:fixture.value})]);',
    'return { selected: fixture.selected, records: values.slice(0,2).map(value =>',
    '  value.status === "fulfilled" ? JSON.parse(value.value).selected : "read-rejected"),',
    '  rejected: values[2].status === "rejected",',
    '  retryReason: values[2].status === "rejected" && values[2].reason.message.includes(fixture.retry), inspection };',
  );
  return lines.join('\n');
}

function onAnyRequest({ request }) {
  const origin = String(request.origin);
  if (!providerActive) rawRequests++;
  if (counters.create >= MAX_REQUESTS || observations.length >= MAX_REQUESTS) {
    observerFailure = 'native-request-limit';

  }
  if (origin !== targetOrigin) {
    foreignRequests++;
    observerFailure = 'unexpected-network-destination';
    void session?.abort().catch(() => {});
  }
}


async function importCompiled() {
  stage = 'compiled-controls-import';
  const symbols = ['createCoreToolRegistry', 'createCodemodeToolDefinition', 'adaptTools',
    'createToolHooks', 'executeToolBatch', 'HookDispatcher', 'inspectImage', 'sessionModelCatalog', 'classifyProviderFailure'];
  const artifacts = buildManifest.artifacts.filter(item => item.path.endsWith('.js')).map(item => {
    const file = path.join(packageRoot, 'dist', item.path), bytes = fs.readFileSync(file);
    check(hash(bytes) === item.sha256, 'compiled-control-artifact-changed');
    return { ...item, file, text: bytes.toString('utf8') };
  });
  const result = {}, owners = [];
  for (const symbol of symbols) {
    const candidates = artifacts.filter(item =>
      new RegExp('(?:function|class|var) ' + symbol + '[ (=]').test(item.text)
      && [...item.text.matchAll(/^export \{([\s\S]*?)^\};/gm)].some(match =>
        match[1].split(',').map(value => value.trim()).includes(symbol)));
    check(candidates.length === 1, 'compiled-control-symbol-unavailable');
    const artifact = candidates[0], imported = await import(pathToFileURL(artifact.file).href);
    check(typeof imported[symbol] === 'function', 'compiled-control-export-unavailable');
    result[symbol] = imported[symbol];
    owners.push({ symbol, artifact: artifact.path, sha256: artifact.sha256 });
  }
  receipt.compiledControls = { manifestBoundInternalExports: true, publicPackageExportClaim: false, owners };
  return result;
}

function selectedOwner(raw) {
  const providers = findNodeAtLocation(parseTree(raw, [], { allowTrailingComma: true }), ['providers']);
  let found, minimal = false;
  for (const property of providers?.children ?? []) {
    const id = property.children?.[0]?.value, node = property.children?.[1];
    if (typeof id !== 'string' || !node) continue;
    let entry;
    try { entry = JSON.parse(raw.slice(node.offset, node.offset + node.length)); }
    catch {
      entry = {};
      for (const field of node.children ?? []) {
        const key = field.children?.[0]?.value, value = field.children?.[1];
        if (!value || !['label', 'runtime', 'endpoint', 'auth', 'baseUrl', 'envVar'].includes(key)) continue;
        try { entry[key] = JSON.parse(raw.slice(value.offset, value.offset + value.length)); } catch {}
      }
      if (entry.runtime === selected.provider && entry.baseUrl?.replace(/\/+$/, '') === selected.baseUrl.replace(/\/+$/, '')
        && entry.endpoint && entry.auth) { entry.models = [{ id: selected.model }]; minimal = true; }
    }
    if ((entry.runtime ?? id) === selected.provider && entry.baseUrl?.replace(/\/+$/, '') === selected.baseUrl.replace(/\/+$/, '')
      && entry.models?.some(model => model.id === selected.model)) {
      check(!found, 'active-configured-catalogue-ambiguous');
      found = { id, ...entry, models: entry.models.filter(model => model.id === selected.model) };
    }
  }
  check(found, 'active-configured-catalogue-unavailable');
  receipt.catalogueSetup = { originalActiveSelected: true, isolatedValidObjects: true,
    minimalConfiguredModelEntry: minimal, originalConfiguredCredentials: true, inventedPrices: false, inventedRemoteLimits: false };
  return found;
}

  receipt.catalogueSetup.localOutputCap = output;
  receipt.catalogueSetup.publishedContextWindow = declared.contextWindow ?? null;
  receipt.catalogueSetup.publishedInputLimit = declared.inputLimit ?? null;
  receipt.catalogueSetup.originalPricesPresent = declared.cost !== undefined;
  receipt.originalInputsSha256 = hash(JSON.stringify([...originalHashes].map(([file, digest]) => [hash(file), digest]).sort()));
  targetOrigin = new URL(selected.baseUrl).origin;
  creating.subscribe(onAnyRequest); creating.subscribe(onCreate); sent.subscribe(onSent);
  headers.subscribe(onHeaders); trailers.subscribe(onTrailers);

try {
  await setup();

    const priorFile = path.join(directory, 'native-network-failed.json');

      && prior.rawCases.length === 6 && prior.rawCases.every(item => item.status === 'passed' && item.zeroProviderRequests),
    'retained-raw-controls-stale');

  receipt.status = 'failed'; receipt.failedStage = stage;
  const detail = String(error?.message ?? '') + ' ' + (lastFailure ?? '');
  const providerTransport = stage.startsWith('provider-');

    : error.probeCode === 'request-stopped-before-dispatch' ? 'local-request-admission-or-harness'

          : providerTransport && /schema|invalid.*param|unsupported.*param/i.test(lastFailure ?? '') ? 'provider-protocol-or-content'
            : error.probeCode?.includes('configured') ? 'configured-target-admission'
              : /actual-native-call|query-unsettled|bounded-native-rounds/.test(error.probeCode ?? '')

  catch { receipt.status = 'failed'; process.exitCode = 1; }
  receipt.privateStateRemoved = !root || !fs.existsSync(root);

      ? hash(fs.readFileSync(path.join(packageRoot, 'dist/build-manifest.json'))) === manifestHash : null;
  } catch { receipt.originalConfigUnchanged = false; }
  if (!receipt.privateStateRemoved || receipt.originalConfigUnchanged === false
    || receipt.sourceUnchanged === false || receipt.buildUnchanged === false) {
    receipt.status = 'failed'; process.exitCode = 1;
  }
  receipt.requests = observations; receipt.observerEvents = counters;
  receipt.actualRequests = observations.length; receipt.foreignRequests = foreignRequests; receipt.rawNetworkRequests = rawRequests;
  fs.writeFileSync(path.join(directory, 'native.json'), JSON.stringify(receipt, null, 2) + '\n', { mode: 0o600 });
  console.log(JSON.stringify({ status: receipt.status, buildId: receipt.buildId,
    failedStage: receipt.failedStage ?? null, failure: receipt.failure ?? null, actualRequests: receipt.actualRequests,
    providerCases: receipt.providerCases.map(item => item.name), rawCases: receipt.rawCases.map(item => item.name),
    privateStateRemoved: receipt.privateStateRemoved, originalConfigUnchanged: receipt.originalConfigUnchanged,
    sourceUnchanged: receipt.sourceUnchanged, buildUnchanged: receipt.buildUnchanged }));
