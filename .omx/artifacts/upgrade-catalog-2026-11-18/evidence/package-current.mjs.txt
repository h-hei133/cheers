function firstJson(text) {

  const start = text.search(/[\[{]/); check(start >= 0, 'pack-json-missing');
  for (let index = start; index < text.length; index++) {

    else if (char === '"') quoted = true;
    else if (char === '{' || char === '[') depth++;
    else if (char === '}' || char === ']') { if (--depth === 0) return JSON.parse(text.slice(start, index + 1)); }
  }

}
async function run(name, args, cwd, ms = 180000) {

  const timer = setTimeout(() => { timedOut = true; child.kill('SIGKILL'); }, ms);
  const append = (data, out) => { bytes += data.length; if (bytes > 8 * 1024 * 1024) { overflow = true; child.kill('SIGKILL'); return; }

  child.stdout.on('data', data => append(data, true)); child.stderr.on('data', data => append(data, false));
  try { const result = await new Promise(resolve => { child.once('error', () => resolve({ exit: null, signal: 'spawn-error' }));
    child.once('close', (exit, signal) => resolve({ exit, signal })); });

  } finally { clearTimeout(timer); }
}
