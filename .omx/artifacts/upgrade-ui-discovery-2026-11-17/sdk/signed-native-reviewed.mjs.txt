function check(condition, code) {
  if (condition) return;
  const error = new Error(code); error.probeCode = code; throw error;
}

function setEnv(key, value) {
  if (!previousEnv.has(key)) previousEnv.set(key, process.env[key]);
  if (value === undefined) delete process.env[key]; else process.env[key] = value;
}

function held() {
  let resolve;
  const promise = new Promise(accept => { resolve = accept; });
  return { promise, resolve };
}

async function bounded(promise, ms, code) {
  let timer;
  try { return await Promise.race([promise, new Promise((_, reject) => {
    timer = setTimeout(() => { const error = new Error(code); error.probeCode = code; reject(error); }, ms);
  })]); } finally { clearTimeout(timer); }
}

async function until(predicate, code, ms = 6000) {
  const end = Date.now() + ms;
  while (!predicate()) {
    check(Date.now() < end, code);
    await new Promise(resolve => setTimeout(resolve, 10));
  }
}

function observeMain({ request }, kind) {
  if (!request) return;
  wire[kind === 'native' ? 'mainNative' : 'mainUndici']++;

  const target = kind === 'native' ? String(request.getHeader?.('host') ?? request.host ?? '') : String(request.origin ?? '');
  if (target !== authority && target !== `http://${authority}`) wire.foreignRequests++;
}

async function command(file, args, cwd = workspace) {
  let output = '', bytes = 0, overflow = false;
  const child = spawn(file, args, { cwd, env: { ...process.env }, stdio: ['ignore', 'pipe', 'pipe'] });
  children.add(child);
  const capture = chunk => {
    bytes += chunk.length;
    if (bytes > 256 * 1024) { overflow = true; child.kill('SIGTERM'); return; }
    output += chunk.toString('utf8');
  };
  child.stdout.on('data', capture); child.stderr.on('data', capture);
  let killTimer;
  const timer = setTimeout(() => {
    child.kill('SIGTERM'); killTimer = setTimeout(() => child.kill('SIGKILL'), 750);
  }, 15_000);
  try {
    const exit = await new Promise(resolve => {
      child.once('error', () => resolve({ code: null, signal: 'spawn_error' }));
      child.once('close', (code, signal) => resolve({ code, signal }));
    });
    check(!overflow, 'native-output-bound');
    return { ...exit, output, bytes, outputSha256: hash(output) };
  } finally { clearTimeout(timer); clearTimeout(killTimer); children.delete(child); }
}

  stage = label;

    '--project', workspace, '--no-context-files', '--no-skills',

  check(result.code === 0 && !result.signal, `${label}-cli-exit`);

  const refused = /(?:Extension|Marketplace) refused:/.test(result.output);

    `${label}-effect-receipt`);
  receipt.lifecycle.push({ code: label, cliExit: result.code, effect: confirmed ? 'confirmed' : 'refused',
    outputBytes: result.bytes, outputSha256: result.outputSha256 });
}

function decoded(observation, expectedTitle, answer, continuation, catalogue) {
  check(observation.kind === 'returned', `${expectedTitle}-service-return`);
  let value;
  try { value = JSON.parse(observation.output.split(/\r?\n/)[0]); } catch { check(false, `${expectedTitle}-worker-output`); }
  check(value.probe === 'E5_SIGNED_UI' && value.factoryAwaited === true && value.title === expectedTitle,
    `${expectedTitle}-async-factory`);
  check(value.answer === answer && value.continuation === continuation, `${expectedTitle}-answer`);
  check(canonical(value.counts) === canonical(catalogue.counts) && typeof value.catalogueText === 'string'
    && hash(value.catalogueText) === catalogue.sha256,
    `${expectedTitle}-real-catalogue`);
  check(value.hasTool && value.activeTool && value.coreRead && value.pluginCommand, `${expectedTitle}-catalogue-consumers`);
  check(observation.output.includes(`Extension execution: completed; exit 0; mode in-process; source ${sourceName}; SHA256 ${receipt.archiveSha256}.`),
    `${expectedTitle}-actual-worker-receipt`);
  return value;
}

    response.writeHead(200, { 'content-type': 'application/json', 'content-length': Buffer.byteLength(body) }); response.end(body);
  });
  server.on('connection', socket => { sockets.add(socket); socket.once('close', () => sockets.delete(socket)); });
  await new Promise(resolve => server.listen(0, '127.0.0.1', resolve)); authority = `127.0.0.1:${server.address().port}`;
  observerFile = path.join(privateRoot, 'observer.mjs'); observerLog = path.join(privateRoot, 'wire.jsonl');
  fs.writeFileSync(observerFile, `import fs from 'node:fs'; import { channel } from 'node:diagnostics_channel';
const file = ${JSON.stringify(observerLog)}, authority = ${JSON.stringify(authority)};
const note = value => fs.appendFileSync(file, JSON.stringify(value) + '\\n', { mode: 0o600 });
note({ kind: 'ready' });
channel('http.client.request.start').subscribe(({request}) => {
  const target = String(request?.getHeader?.('host') ?? request?.host ?? '');
  note({ kind: 'native', post: String(request?.method).toUpperCase() === 'POST', foreign: target !== authority });
});
channel('undici:request:create').subscribe(({request}) => {
  note({ kind: 'undici', post: String(request?.method).toUpperCase() === 'POST', foreign: String(request?.origin) !== 'http://' + authority });
});
`, { mode: 0o600 });
  nativeChannel.subscribe(onNative); undiciChannel.subscribe(onUndici);
  await cli('source-add', `/marketplace add ${sourceName} http://${authority}/registry`, 'confirmed');
  await cli('source-trust', `/marketplace trust ${sourceName} ${keyId} publisher.pem`, 'confirmed');
  const source = JSON.parse(fs.readFileSync(sourcesFile, 'utf8')).find(entry => entry.name === sourceName);
  check(source?.trust?.fingerprint === receipt.trustFingerprint && source.trust.keyId === keyId
    && source.trust.publicKey === publicPem && source.trust.provenance.startsWith('explicit-user-file:'), 'actual-explicit-trust');
  check(state().packages.length === 0 && serverEvents.length === 0, 'registration-does-not-install');
  envelopeMode = 'tampered';
  await cli('bad-signature', `/plugin install ${sourceName}:${packageName}@1.0.0`, /Extension refused: Untrusted registry signature\./);
  check(state().packages.length === 0 && !serverEvents.some(entry => entry.kind === 'archive'), 'signature-refusal-before-archive');
  envelopeMode = 'valid'; archiveMode = 'tampered';
  await cli('bad-archive', `/plugin install ${sourceName}:${packageName}@1.0.0`, /Extension archive integrity mismatch or size limit\./);
  check(state().packages.length === 0 && !fs.existsSync(path.join(path.dirname(registryFile), 'packages', packageName, receipt.archiveSha256)),
    'bad-archive-not-installed');
  archiveMode = 'valid';

  check(installed && installed.active.digest === receipt.archiveSha256 && installed.active.source === sourceName
    && installed.active.sourceIdentity === hash(canonical(source)) && installed.active.trustFingerprint === receipt.trustFingerprint
    && canonical(installed.active.manifest) === canonical(manifest) && installed.active.storage.length > 0
    && installed.active.files['scripts/worker.ts'] === receipt.workerSourceSha256
    && Object.keys(installed.activation).length === 0, 'signed-installed-provenance');
  check(hash(fs.readFileSync(path.join(path.dirname(registryFile), 'packages', packageName, receipt.archiveSha256, 'scripts/worker.ts')))
    === receipt.workerSourceSha256, 'downloaded-installed-worker-bytes');

  check(state().packages[0].activation[fs.realpathSync(workspace)].enabled
    && state().packages[0].activation[fs.realpathSync(workspace)].grants.length === 0, 'enable-does-not-grant');

  await cli('tool-grant', `/plugin grant ${packageName} tool:worker_ui`, 'confirmed');
  const activation = state().packages[0].activation[fs.realpathSync(workspace)];
  check(activation.enabled && canonical([...activation.grants].sort()) === canonical(['script:main', 'tool:worker_ui']), 'actual-separate-grants');
  receipt.signedInstallation = true; receipt.sourceIdentity = installed.active.sourceIdentity;
  receipt.inventoryFiles = Object.keys(installed.active.files).length;
  setEnv('CHEERS_OFFLINE', '1');
  sdk = await import(pathToFileURL(path.join(repo, 'packages/ts-shell', pkg.exports['.'].import)).href);
  ({ ExtensionService } = await import(pathToFileURL(path.join(repo, 'packages/ts-shell', pkg.exports['./extensions'].import)).href));

    && typeof ExtensionService === 'function', 'public-sdk-exports');
  receipt.publicCompiledSdk = true;
  await runMode('options.ui');
  await runMode('bindUi');
  receipt.status = 'passed';
} catch (error) {
  receipt.status = 'failed'; receipt.failure = { stage, code: error?.probeCode ?? 'unexpected-error' }; process.exitCode = 1;
} finally {

  if (observerLog && fs.existsSync(observerLog)) {

  }

    .catch(() => { receipt.status = 'failed'; receipt.cleanupFailure = 'registry-server'; process.exitCode = 1; });
  for (const [key, value] of previousEnv) { if (value === undefined) delete process.env[key]; else process.env[key] = value; }

    receipt.personalInputsUnchanged = [...personalInputs].every(([file, before]) => (fs.existsSync(file) ? hash(fs.readFileSync(file)) : null) === before);
    receipt.personalInputsChecked = personalInputs.size;
    receipt.sourceUnchanged = sourceBefore !== undefined && canonical(sourceHashes()) === canonical(sourceBefore);
    if (privateRoot) fs.rmSync(privateRoot, { recursive: true, force: true });
    receipt.privateStateRemoved = privateRoot === undefined || !fs.existsSync(privateRoot);
  } catch { receipt.privateStateRemoved = false; }
  if (!receipt.personalInputsUnchanged || !receipt.sourceUnchanged || !receipt.privateStateRemoved) {
    receipt.status = 'failed'; receipt.cleanupFailure ??= 'private-state-or-input-identity'; process.exitCode = 1;
  }
  if (receiptFile.startsWith(`${path.join(repo, '.tmp')}${path.sep}`)) {
    fs.mkdirSync(path.dirname(receiptFile), { recursive: true });
    fs.writeFileSync(receiptFile, `${JSON.stringify(receipt, null, 2)}\n`, { mode: 0o600 });
  }
  process.stdout.write(`${JSON.stringify(receipt, null, 2)}\n`);
}
