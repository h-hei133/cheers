async function until(predicate, label, timeout = 20_000) {
  const end = Date.now() + timeout;
  for (;;) { if (predicate()) return; if (Date.now() > end) throw new Error(`Deadline: ${label}`); await delay(25); }
}

}
function start(fixture, approval = 'allow-once', sessionId) {
  const responses = [], frames = [], waiters = new Set(); let lines = '', invalidLines = 0;
  const env = { ...process.env, HOME: fixture.home, USERPROFILE: fixture.home, CHEERS_TEST_ROOT: fixture.home,
    GIT_CEILING_DIRECTORIES: root, CHEERS_PROVIDER_MODE: 'real', CHEERS_REAL_PROVIDER_SMOKE: '', OPENAI_API_KEY: '', ANTHROPIC_API_KEY: '',
    CHEERS_MARKETPLACES_FILE: path.join(fixture.home, 'sources.json'), CHEERS_EXTENSIONS_FILE: path.join(fixture.home, 'extensions.json') };
  const spawned = spawn('rtk', ['node', path.join(repo, 'scripts/start.mjs'), '--rpc', '--project', fixture.workspace,
    '--no-context-files', '--no-skills', '--tools', 'bash,read', '--model', 'openai/rpc-retry',
    '--non-interactive-approval', approval, ...(sessionId ? ['--session-id', sessionId] : [])],
    { cwd: fixture.workspace, env, stdio: ['pipe', 'pipe', 'pipe'] });
  const closed = new Promise(resolve => spawned.on('close', (code, signal) => resolve({ code, signal })));
  spawned.stderr.on('data', () => {});
  spawned.stdout.on('data', chunk => {
    lines += chunk.toString();
    for (;;) {
      const newline = lines.indexOf('\n'); if (newline < 0) break;
      const line = lines.slice(0, newline); lines = lines.slice(newline + 1);
      if (!line.trim()) continue;
      let value; try { value = JSON.parse(line); } catch { invalidLines++; continue; }
      frames.push(value); if (value.type === 'response') responses.push(value);
      for (const waiter of [...waiters]) if (waiter.predicate(value)) waiter.resolve(value);
    }
  });
  function waitFor(predicate, from = frames.length) {
    const found = frames.slice(from).find(predicate); if (found) return Promise.resolve(found);
    return new Promise((resolve, reject) => {
      const timer = setTimeout(() => { waiters.delete(waiter); reject(new Error('RPC response deadline.')); }, 30_000);
      const waiter = { predicate, resolve: value => { clearTimeout(timer); waiters.delete(waiter); resolve(value); } };
      waiters.add(waiter);
    });
  }
  function command(type, values = {}) {
    const id = String(++number), response = waitFor(value => value.type === 'response' && value.id === id);
    spawned.stdin.write(JSON.stringify({ id, type, ...values }) + '\n'); return response;
  }
  const client = { ...fixture, process: spawned, closed, frames, command, waitFor, get invalidLines() { return invalidLines; },
    async finish() {
      spawned.stdin.end(); const result = await closed; clients.delete(client);
      assert.deepEqual(result, { code: 0, signal: null }); assert.equal(invalidLines, 0); assert.equal(lines.trim(), '');
      return result;
    } };
  clients.add(client); return client;
}
function transcript(client, id) {
  return fs.readFileSync(path.join(client.storage, 'sessions', id, 'transcript.jsonl'), 'utf8').trim().split('\n').map(JSON.parse);
}

let baseUrl, mode = 'ordinary', held, accepted = false, refused = false, streamClosed = false;
const observed = [];
function answer(response, text) {
  response.writeHead(200, { 'content-type': 'text/event-stream' });
  response.end(`data: ${JSON.stringify({ choices: [{ delta: { content: text }, finish_reason: null }] })}\n\ndata: ${JSON.stringify({ choices: [{ delta: {}, finish_reason: 'stop' }], usage: { prompt_tokens: 24, completion_tokens: 8 } })}\n\ndata: [DONE]\n\n`);
}
try {
  assert.equal(process.platform, 'linux');
  server = http.createServer(async (request, response) => {
    const chunks = []; for await (const chunk of request) chunks.push(Buffer.from(chunk));
    if (request.method !== 'POST') { response.writeHead(200, { 'content-type': 'application/json' }); response.end('{"data":[]}'); return; }
    const body = JSON.parse(Buffer.concat(chunks).toString());
    requestBodies.push(body); bodyCount++; observed.push({ mode, accepted: mode === 'ordinary' || mode === 'stream' || mode === 'success' });
    if (mode === 'ordinary' || mode === 'stream') {
      held = response; accepted = true;
      response.writeHead(200, { 'content-type': 'text/event-stream' }); response.flushHeaders();
      response.once('close', () => { streamClosed = true; });
      return;
    }
    if (mode === 'provider' || mode === 'outer') {
      response.once('finish', () => { refused = true; });
      response.writeHead(429, { 'content-type': 'application/json', 'retry-after': '10' });
      response.end('{"error":{"type":"rate_limit_error","message":"local rate limit"}}');
      return;
    }
    answer(response, 'RPC_RETRY_SUCCESS');
  });
  await new Promise(resolve => server.listen(0, '127.0.0.1', resolve));
  baseUrl = `http://127.0.0.1:${server.address().port}`;

  stage = 'ordinary'; current = start(setup('ordinary', { retryEnabled: true, retryMaxRetries: 2, providerRetryMaxRetries: 2 }));
  assert.equal((await current.command('get_state')).success, true);
  let end = current.waitFor(frame => frame.type === 'agent_end');
  assert.equal((await current.command('prompt', { message: 'ordinary request' })).success, true);
  await until(() => accepted, 'accepted ordinary stream');
  assert.equal((await current.command('abort_retry')).success, true);
  assert.equal(streamClosed, false);
  held.end(`data: ${JSON.stringify({ choices: [{ delta: { content: 'ORDINARY_STILL_LIVE' }, finish_reason: null }] })}\n\ndata: ${JSON.stringify({ choices: [{ delta: {}, finish_reason: 'stop' }] })}\n\ndata: [DONE]\n\n`);
  assert.equal((await end).status, 'completed');
  assert.equal(observed.filter(value => value.mode === 'ordinary').length, 1);
  receipt.stages.ordinary = { passed: true, acceptedStreamSurvived: true, completed: true, singlePost: true };
  await current.finish();

  for (const kind of ['provider', 'outer']) {
    stage = kind; mode = kind; refused = false;
    current = start(setup(kind, { retryEnabled: true, retryMaxRetries: 2, providerRetryMaxRetries: kind === 'provider' ? 2 : 0,
      retryBaseDelayMs: 10000, retryMaxDelayMs: 10000 }));
    assert.equal((await current.command('get_state')).success, true);
    end = current.waitFor(frame => frame.type === 'agent_end');
    assert.equal((await current.command('prompt', { message: kind + ' backoff request' })).success, true);
    await until(() => refused, kind + ' refusal delivered'); await delay(200);
    assert.equal((await current.command('get_state')).data.isStreaming, true);
    assert.equal((await current.command('abort_retry')).success, true);
    const outcome = await end;
    assert.equal(outcome.status, 'failed'); assert.match(outcome.error, /Retry was cancelled/);
    assert.equal(observed.filter(value => value.mode === kind).length, 1);
    receipt.stages[kind] = { passed: true, failedOutcome: true, singlePost: true, noSecondAttempt: true };
    stage = kind + '-reuse'; mode = 'success';
    end = current.waitFor(frame => frame.type === 'agent_end');
    assert.equal((await current.command('prompt', { message: 'explicit reuse after ' + kind })).success, true);
    assert.equal((await end).status, 'completed');
    receipt.stages[kind].explicitReuseCompleted = true;
    await current.finish();
  }

  stage = 'command'; mode = 'success';
  current = start(setup('command'));
  assert.equal((await current.command('get_state')).success, true);
  const entered = path.join(current.workspace, 'entered'), release = path.join(current.workspace, 'release');
  const shell = current.command('bash', { command: "touch entered; while test ! -f release; do sleep 0.02; done; printf RPC_COMMAND_SURVIVED" });
  await until(() => fs.existsSync(entered), 'native shell entered');
  assert.equal((await current.command('abort_retry')).success, true); fs.writeFileSync(release, 'release');
  const shellResult = await shell;
  assert.equal(shellResult.success, true); assert.equal(shellResult.data.exitCode, 0); assert.match(shellResult.data.output, /RPC_COMMAND_SURVIVED/);
  receipt.stages.command = { passed: true, nativeCommandSurvived: true, exit0: true };
  await current.finish();

  stage = 'accepted-stream'; mode = 'stream'; accepted = false; streamClosed = false;
  current = start(setup('accepted-stream', { retryEnabled: true, retryMaxRetries: 2, providerRetryMaxRetries: 2 }));
  assert.equal((await current.command('get_state')).success, true);
  end = current.waitFor(frame => frame.type === 'agent_end');
  assert.equal((await current.command('prompt', { message: 'accepted response must never replay' })).success, true);
  await until(() => accepted, 'accepted response');
  assert.equal((await current.command('abort_retry')).success, true); assert.equal(streamClosed, false);
  held.destroy();
  const failed = await end;
  assert.equal(failed.status, 'failed'); assert.doesNotMatch(failed.error, /Retry was cancelled/);
  assert.equal(observed.filter(value => value.mode === 'stream').length, 1);

  await current.finish();

  receipt.posts = bodyCount; receipt.status = 'passed'; receipt.validJsonLines = true; receipt.eofExit0 = true;
} catch(error) {
  receipt.status = 'failed'; receipt.failedStage = stage; receipt.posts = bodyCount;
  receipt.failure = { name: error?.name, code: error?.code, message: 'Local retry probe failed; see retained assertions.' };
  receipt.assertion = { stackFrame: error?.stack?.split('\n').find(line => line.includes('retry-native.mjs:')) };
  process.exitCode = 1;
} finally {
  clearTimeout(watchdog); held?.destroy(); server?.closeAllConnections();
  for (const client of clients) { client.process.kill('SIGTERM'); await Promise.race([client.closed, delay(5000)]); }
  await new Promise(resolve => server ? server.close(resolve) : resolve());
  fs.writeFileSync(path.join(directory, 'retry.json'), JSON.stringify(receipt, null, 2) + '\n');
  console.log(JSON.stringify(receipt)); fs.rmSync(root, { recursive: true, force: true });
}
