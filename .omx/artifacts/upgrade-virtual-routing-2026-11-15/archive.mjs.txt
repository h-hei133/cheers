const archive = path.join(repo, '.omx/artifacts/upgrade-virtual-routing-2026-11-15');
const hash = value => createHash('sha256').update(value).digest('hex');
const git = (args, cwd = repo) => {
  const result = spawnSync('rtk', ['proxy', 'git', ...args], { cwd, encoding: 'utf8', maxBuffer: 16 * 1024 * 1024 });
  assert.equal(result.status, 0, 'Git evidence read failed');
  return result.stdout;
};
const list = args => git(args).split('\0').filter(Boolean);

for (const name of receipts) {

  assert.equal(receipt.status, 'passed', name);
  assert.equal(receipt.buildId, build.buildId, name);
  assert.equal(receipt.probeSha256, hash(fs.readFileSync(path.join(scratch, name + '.mjs'))), name);
}

  assert.equal(reversal.sha256, hash(fs.readFileSync(path.join(repo, 'packages/ts-shell/src', reversal.file))));
}

const productFiles = [...new Set([
  ...list(['diff', '--name-only', '-z', '--', 'packages/ts-shell/src']),
  ...list(['ls-files', '--others', '--exclude-standard', '-z', '--', 'packages/ts-shell/src']),
])].sort();
const sources = {};
for (const file of productFiles) {
  const bytes = fs.readFileSync(path.join(repo, file));
  const snapshot = 'source/' + file.replace('packages/ts-shell/src/', '') + '.txt';
  const target = path.join(archive, snapshot);
  fs.mkdirSync(path.dirname(target), { recursive: true });
  fs.writeFileSync(target, bytes);
  sources[file] = { snapshot, sha256: hash(bytes) };
}

for (const entry of fs.readdirSync(scratch, { withFileTypes: true })) {
  if (!entry.isFile() || !/\.(?:json|log|mjs|mts|txt)$/.test(entry.name)) continue;
  const bytes = fs.readFileSync(path.join(scratch, entry.name));
  const text = bytes.toString('utf8');
  assert.ok(!/\b(?:sk|rk|ghp|github_pat)-[A-Za-z0-9_-]{24,}\b|Bearer [A-Za-z0-9_.-]{24,}|-----BEGIN [A-Z ]*PRIVATE KEY-----|https?:\/\/[^\s/:]+:[^\s/@]+@/.test(text), 'Credential pattern found in ' + entry.name);
  const name = /\.(?:mjs|mts)$/.test(entry.name) ? entry.name + '.txt' : entry.name;
  fs.writeFileSync(path.join(archive, name), bytes);
}

const lanes = {};
for (const name of ['virtual-input', 'virtual-selection']) {
  const cwd = path.join(repo, '.tmp', name);
  if (!fs.existsSync(cwd)) continue;
  const changed = git(['diff', '--binary', '--', 'packages/ts-shell/src'], cwd);
  fs.writeFileSync(path.join(archive, name + '-lane.patch.txt'), changed);
  const untracked = git(['ls-files', '--others', '--exclude-standard', '-z', '--', 'packages/ts-shell/src'], cwd).split('\0').filter(Boolean);
  const laneSources = {};
  for (const file of untracked) {
    const bytes = fs.readFileSync(path.join(cwd, file));
    const snapshot = 'lanes/' + name + '/' + file.replace('packages/ts-shell/src/', '') + '.txt';
    fs.mkdirSync(path.dirname(path.join(archive, snapshot)), { recursive: true });
    fs.writeFileSync(path.join(archive, snapshot), bytes);
    laneSources[file] = { snapshot, sha256: hash(bytes) };
  }
  lanes[name] = { patch: name + '-lane.patch.txt', sourceSnapshots: laneSources, integrated: true,
    note: 'Historical lane bytes; integrated current source snapshots are authoritative. Do not recopy.' };
}
fs.writeFileSync(path.join(archive, 'build-manifest.json'), JSON.stringify(build, null, 2) + '\n');
const pack = JSON.parse(fs.readFileSync(path.join(scratch, 'pack.json')))['@cheers/ts-shell'];
const evidence = {};
function collect(directory, relative = '') {
  for (const entry of fs.readdirSync(directory, { withFileTypes: true })) {
    const name = path.join(relative, entry.name);
    if (entry.isDirectory()) collect(path.join(directory, entry.name), name);
    else if (entry.name !== 'delivery.json') evidence[name] = hash(fs.readFileSync(path.join(directory, entry.name)));
  }
}
