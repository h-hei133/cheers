import fs from 'node:fs';
import assert from 'node:assert/strict';
import { createHash } from 'node:crypto';
import { spawnSync } from 'node:child_process';
const base = 'packages/ts-shell/src/cli/rpc/';
const cases = [
  { name: 'user-rows', file: 'host/read.ts', before: 'point.prompt !== undefined && !point.unavailable',
    after: '!point.unavailable', filter: 'offers only saved user questions' },
  { name: 'fork-draft', file: 'host/session.ts', before: 'return selected.cancelled ? selected : { ...selected, text: point.prompt };',
    after: 'return selected;', filter: 'returns a saved question as a fork draft' },
  { name: 'source-version', file: 'host/session.ts', before: 'entryId, expectedVersion: snapshot.version',
    after: 'entryId', filter: 'binds a fork draft' },
  { name: 'fork-consumer-admission', file: 'host/session.ts', before: '      snapshot.assertCurrent();',
    after: '      /* reversed consumer admission */', filter: 'refuses fork when selection changes after the nested read resolves' },
  { name: 'list-consumer-admission', file: 'host/read.ts', before: '  snapshot.assertCurrent();',
    after: '  /* reversed consumer admission */', filter: 'refuses get_fork_messages when selection changes after the nested read resolves' },
];
const receipts = [];
for (const item of cases) {
  const file = base + item.file, original = fs.readFileSync(file), source = original.toString();
  assert.equal(source.split(item.before).length, 2);
  let result;
  try {
    fs.writeFileSync(file, source.replace(item.before, item.after));
    result = spawnSync('rtk', ['env', 'VITEST_MAX_THREADS=2', 'VITEST_MIN_THREADS=2', 'node_modules/.bin/vitest',
      'run', '--root', 'packages/ts-shell', 'src/cli/rpc/runtimeHost.test.ts', '--maxWorkers=1', '--no-file-parallelism', '-t', item.filter],
      { encoding: 'utf8', timeout: 55_000, maxBuffer: 1024 * 1024 });
  } finally { fs.writeFileSync(file, original); }
  const output = (result.stdout ?? '') + (result.stderr ?? '');
  fs.writeFileSync('.tmp/rpc-commands/reversal-' + item.name + '.log', output);
  const receipt = { name: item.name, file, exitCode: result.status, signal: result.signal,
    assertionFailure: result.status === 1 && output.includes('AssertionError'), restoredExact: fs.readFileSync(file).equals(original),
    sourceSha256: createHash('sha256').update(original).digest('hex') };
  receipts.push(receipt); fs.writeFileSync('.tmp/rpc-commands/reversals.json', JSON.stringify(receipts, null, 2) + '\n');
  assert.ok(receipt.assertionFailure); assert.ok(receipt.restoredExact); console.log(JSON.stringify(receipt));
}
