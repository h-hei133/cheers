function wireTools(body) {
  const tools = [];
  const add = (id, content) => {
    const text = textOf(content);
    tools.push({ id, digest: hash(text), bytes: Buffer.byteLength(text), middle: text.includes(MIDDLE),
      pruned: text.includes('[... Omitted') || text.includes('Duplicate tool output') });
  };
  for (const message of body.messages ?? []) {
    if (message.role === 'tool') add(message.tool_call_id, message.content);
    for (const block of Array.isArray(message.content) ? message.content : []) {
      if (block.type === 'tool_result') add(block.tool_use_id, block.content);
      if (block.toolResult) add(block.toolResult.toolUseId, block.toolResult.content);
    }
  }
  for (const item of Array.isArray(body.input) ? body.input : []) {
    if (item.type === 'function_call_output') add(item.call_id, item.output);
  }
  for (const content of body.contents ?? []) for (const part of content.parts ?? []) {
    if (part.functionResponse) add(part.functionResponse.id, part.functionResponse.response?.output);
  }
  return tools;
}

// Keep raw chunks only until completion; headers, credentials and bodies never reach a receipt.
function responseUsage(bytes, encoding) {
  const decompress = encoding === 'gzip' ? gunzipSync : encoding === 'deflate' ? inflateSync
    : encoding === 'br' ? brotliDecompressSync : undefined;
  const body = (decompress ? decompress(bytes, { maxOutputLength: MAX_BYTES }) : bytes).toString('utf8');
  const usage = {};
  let parsed = 0;
  const capture = frame => {
    parsed++;
    const value = frame?.response?.usage ?? frame?.message?.usage ?? frame?.usage;
    if (value && typeof value === 'object') {
      for (const [field, candidates] of Object.entries({
        input: [value.prompt_tokens, value.input_tokens],
        output: [value.completion_tokens, value.output_tokens],
        total: [value.total_tokens],
        read: [value.cache_read_input_tokens, value.prompt_tokens_details?.cached_tokens, value.input_tokens_details?.cached_tokens],
        write: [value.cache_creation_input_tokens],
      })) {
        const found = candidates.find(reported);
        if (found !== undefined) usage[field] = found;
      }
    }
    const google = frame?.usageMetadata;
    if (google) for (const [field, name] of Object.entries({
      input: 'promptTokenCount', output: 'candidatesTokenCount', total: 'totalTokenCount', read: 'cachedContentTokenCount',
    })) if (reported(google[name])) usage[field] = google[name];
  };
  try { const frame = JSON.parse(body); for (const value of Array.isArray(frame) ? frame : [frame]) capture(value); }
  catch {
    for (const line of body.split(/\r?\n/)) {
      if (!line.startsWith('data:')) continue;
      const value = line.slice(5).trim(); if (!value || value === '[DONE]') continue;
      try { capture(JSON.parse(value)); } catch { /* No raw malformed event is recorded. */ }
    }
  }
  return { inspectable: parsed > 0, input: usage.input ?? null, output: usage.output ?? null, total: usage.total ?? null,
    read: usage.read ?? 0, write: usage.write ?? 0, readReported: reported(usage.read), writeReported: reported(usage.write) };
}

const onCreate = ({ request }) => {
  if (request.method !== 'POST' || String(request.origin) !== targetOrigin) return;
  observerEvents.create++;
  const capture = { stage, outgoing: [], incoming: [], outgoingBytes: 0, incomingBytes: 0 };
  pendingBodies.set(request, capture);
  const originalBody = request.onBodySent, originalData = request.onData;
  const originalHeaders = request.onHeaders, originalComplete = request.onComplete;
  const originalError = request.onError;
  request.onBodySent = function (chunk) {
    try {
      capture.outgoingBytes += Buffer.byteLength(chunk);
      if (capture.outgoingBytes <= MAX_BYTES) capture.outgoing.push(Buffer.from(chunk));
      else { observerFailure = 'request-capture-limit'; capture.outgoing.length = 0; }
    } catch { observerFailure = 'request-capture'; }
    return originalBody.call(this, chunk);
  };
  request.onData = function (chunk) {
    try {
      observerEvents.data++;
      capture.incomingBytes += Buffer.byteLength(chunk);
      if (capture.incomingBytes <= MAX_BYTES) capture.incoming.push(Buffer.from(chunk));
      else { observerFailure = 'response-capture-limit'; capture.incoming.length = 0; }
    } catch { observerFailure = 'response-capture'; }
    return originalData.call(this, chunk);
  };
  request.onHeaders = function (statusCode, rawHeaders, ...rest) {
    onHeaders({ request: this, response: { statusCode, headers: rawHeaders } });
    return originalHeaders.call(this, statusCode, rawHeaders, ...rest);
  };
  request.onComplete = function (...args) {
    observerEvents.complete++;
    onTrailers({ request: this });
    return originalComplete.apply(this, args);
  };
  request.onError = function (...args) {
    observerEvents.retired++;
    const captured = pendingBodies.get(this);
    if (captured?.statusCode) {
      captured.endedBy = 'accepted-stream-retirement';
      onTrailers({ request: this });
    }
    return originalError.apply(this, args);
  };
};
const onSent = ({ request }) => {
  const capture = pendingBodies.get(request); if (!capture) return;
  try {
    check(observations.length < MAX_REQUESTS, 'native-request-limit');
    check(capture.outgoingBytes <= MAX_BYTES, 'request-capture-limit');
    const bytes = Buffer.concat(capture.outgoing), body = JSON.parse(bytes.toString('utf8'));
    capture.observation = { stage: capture.stage, bodySent: true, bytes: capture.outgoingBytes,
      digest: hash(bytes), checkpoint: bytes.includes(Buffer.from('[Context Checkpoint]')), tools: wireTools(body) };
    observations.push(capture.observation);
  } catch (error) { observerFailure = error.probeCode ?? 'request-capture'; void session?.abort().catch(() => {}); }
  finally { capture.outgoing.length = 0; }
};
const onHeaders = ({ request, response }) => {
  const capture = pendingBodies.get(request); if (!capture) return;
  observerEvents.headers++;
  capture.statusCode = response.statusCode;
  for (let index = 0; index < response.headers.length; index += 2) {
    if (String(response.headers[index]).toLowerCase() === 'content-encoding')
      capture.encoding = String(response.headers[index + 1]).toLowerCase();
  }
};
const onTrailers = ({ request }) => {
  const capture = pendingBodies.get(request); if (!capture) return;
  observerEvents.trailers++;
  try {
    check(capture.incomingBytes <= MAX_BYTES, 'response-capture-limit');
    if (capture.observation) Object.assign(capture.observation, {
      responseStatus: capture.statusCode, usage: responseUsage(Buffer.concat(capture.incoming), capture.encoding),
      responseBytes: capture.incomingBytes,
      responseEnd: capture.endedBy ?? 'transport-complete',
    });
  } catch { observerFailure = 'response-capture'; }
  finally { capture.incoming.length = 0; pendingBodies.delete(request); }
};

function verifyAccounting(rows, requests) {
  check(rows.length === requests.length, 'accounting-request-correlation');
  for (const [index, row] of rows.entries()) {
    const raw = requests[index].usage;
    check(raw?.inspectable, 'response-usage-unobservable');
    check(typeof row.cacheReadReported === 'boolean' && typeof row.cacheWriteReported === 'boolean',
      'cache-category-presence-missing');
    check(row.cacheReadReported === raw.readReported && row.cacheWriteReported === raw.writeReported,
      'cache-category-presence-mismatch');
    check(row.cacheReadTokens === raw.read && row.cacheWriteTokens === raw.write, 'cache-counter-mismatch');
    if (row.attribution) {
      check(row.attribution.provider === selected.provider && row.attribution.model === selected.model, 'usage-target-mismatch');
      equal(row.attribution.rates, rates, 'captured-prices-mismatch');
    }
    if (row.kind === 'request') {
      check(row.attribution && raw.input !== null, 'request-measurement-missing');
      check(typeof row.id === 'string' && Number.isSafeInteger(row.timestamp), 'request-persistence-identity');
    } else check(row.kind === 'usage-gap', 'unexpected-accounting-boundary');
  }
  return rows.map(row => ({ kind: row.kind, promptTokens: row.promptTokens, completionTokens: row.completionTokens ?? null,
    cacheReadTokens: row.cacheReadTokens, cacheWriteTokens: row.cacheWriteTokens,
    cacheReadReported: row.cacheReadReported, cacheWriteReported: row.cacheWriteReported,
    capturedRates: row.attribution ? row.attribution.rates : null }));
}

async function prompt(name, input, { expectRead = false, accounting = true } = {}) {
  stage = name; lastFailure = undefined;
  const first = observations.length, beforeUsage = savedUsage()?.recentRequests ?? [], beforeTools = sourceResults().length;
  let completed = false, unproven = false, successfulRead = 0, toolStarts = 0;
  const unsubscribe = session.subscribe(event => {
    if (event.type === 'query.completed') completed = true;
    if (event.type === 'query.completed_unproven') unproven = true;
    if (event.type === 'query.failed' || event.type === 'lifecycle.error') lastFailure = String(event.error);
    if (event.type === 'tool.started') toolStarts++;
    if (event.type === 'tool.result' && event.toolName === 'read' && event.success) successfulRead++;
  });
  let result;
  try { result = await session.prompt(input); } finally { unsubscribe(); }
  check(!observerFailure, observerFailure ?? 'wire-observer');
  check(!receipt.deadline, 'native-deadline');
  check(!lastFailure && !unproven && (completed || result.includes(ACK)), 'query-not-completed');
  const requests = observations.slice(first);
  check(requests.length > 0 && requests.every(request => request.bodySent), 'native-dispatch-missing');
  if (expectRead) {
    receipt.readAttempts ??= [];
    receipt.readAttempts.push({ name, toolStarts, successfulRead, newSourceResults: sourceResults().length - beforeTools });
    check(toolStarts === 1 && successfulRead === 1 && sourceResults().length === beforeTools + 1, 'actual-read-not-observed');
  }
  else check(toolStarts === 0 && successfulRead === 0, 'unexpected-tool-during-continuation');
  const beforeIds = new Set(beforeUsage.map(entry => entry.id));
  const rows = (savedUsage()?.recentRequests ?? []).filter(entry => !beforeIds.has(entry.id));
  const usage = accounting ? verifyAccounting(rows, requests) : undefined;
  receipt.cases.push({ name, passed: true, actualRequests: requests.length, successfulRead, toolStarts, acknowledged: result.includes(ACK),
    completed, seededHistory: name === 'automatic-compaction', ...(usage ? { usage } : {}) });
  return requests;
}

  const commitsBefore = transcript().filter(event => event.type === 'compaction-commit').length;
  for (let index = 0; index < pairs; index++) {
    const facts = ('Seeded bounded observation ' + index + ': preserve the acknowledged source-history decision. ').repeat(20).slice(0, 1200);
    session.sessionManager.addTurn(id, { role: 'user', content: 'Recorded fixture work ' + index + ': ' + facts });
    session.sessionManager.addTurn(id, { role: 'assistant', content: 'Recorded fixture result ' + index + ': ' + facts });
  }
  receipt.compactionFixture = { seededPairs: pairs, seededMessages: pairs * 2, localWindow: pressureWindow,
    keepRecentTokens: 600, inventedRemoteLimits: false, seededMessagesAreProviderRounds: false };
  const automatic = await prompt('automatic-compaction', 'Continue with a brief acknowledgement and no tools.', { accounting: false });
  const commitsAfter = transcript().filter(event => event.type === 'compaction-commit').length;
  check(commitsAfter > commitsBefore && automatic.some(request => request.checkpoint), 'automatic-compaction-not-published');
  const storedTools = transcript().filter(event => event.type === 'message' && event.role === 'tool'
    && originalTools.some(tool => tool.id === event.id));
  check(storedTools.length === 2 && storedTools.every(event => originalTools.find(tool => tool.id === event.id).content === event.content),
    'compaction-mutated-original-tool-source');
  receipt.cases.push({ name: 'automatic-publication', passed: true, actualCommitIncrease: commitsAfter - commitsBefore,
    checkpointDispatched: true, seededPressure: true });
  const afterCompactionUsage = structuredClone(savedUsage());
  await close(); await open(pressureConfig, id);
  equal(savedUsage(), afterCompactionUsage, 'compaction-reopen-usage-mismatch');
  const continued = await prompt('reopened-checkpoint', 'Acknowledge the continued session without tools.');
  check(continued.some(request => request.checkpoint), 'reopened-checkpoint-not-dispatched');
  await close();
  check(JSON.parse(fs.readFileSync(path.join(repo, 'packages/ts-shell/dist/build-manifest.json'), 'utf8')).buildId === receipt.buildId,
    'build-changed-during-probe');
  receipt.realPromptRounds = receipt.cases.filter(item => typeof item.actualRequests === 'number').length;
  receipt.cacheClaim = 'Only observed category presence and recorded counters; no savings, TTL, cache-control or remote-capacity claim.';
  receipt.status = 'passed';
} catch (error) {
  receipt.status = 'failed'; receipt.failedStage = stage;
  const detail = String(error?.message ?? '') + ' ' + (lastFailure ?? '');
  receipt.failure = { category: receipt.deadline ? 'deadline' : /429|quota|rate.?limit/i.test(detail) ? 'provider-quota'
    : /401|403|auth|credential/i.test(detail) ? 'provider-auth' : /ECONN|EHOST|ENOTFOUND|fetch failed|network|timeout/i.test(detail) ? 'provider-network'
      : error.probeCode?.includes('configured') ? 'configured-target-admission' : 'product-or-probe-contract',
    code: error.probeCode ?? 'native-operation-failed' };
  receipt.failureLine = String(error?.stack ?? '').match(/native\.mjs:(\d+)/)?.[1] ?? null;
  process.exitCode = 1;
} finally {
  clearTimeout(deadline);
  try { await close(); } catch { receipt.cleanupSettled = false; receipt.status = 'failed'; process.exitCode = 1; }
  creating.unsubscribe(onCreate); sent.unsubscribe(onSent); headers.unsubscribe(onHeaders); trailers.unsubscribe(onTrailers);
  for (const [key, value] of previousEnv) { if (value === undefined) delete process.env[key]; else process.env[key] = value; }
  try { if (root) fs.rmSync(root, { recursive: true, force: true }); }
  catch { receipt.status = 'failed'; process.exitCode = 1; }
  receipt.privateStateRemoved = !root || !fs.existsSync(root);
  try {
    receipt.originalConfigUnchanged = originalHashes.size > 0 ? [...originalHashes].every(([file, before]) =>
      (fs.existsSync(file) ? hash(fs.readFileSync(file)) : null) === before) : null;
  } catch { receipt.originalConfigUnchanged = false; }
  if (!receipt.privateStateRemoved || receipt.originalConfigUnchanged === false) { receipt.status = 'failed'; process.exitCode = 1; }
  receipt.requests = observations.map(request => ({ stage: request.stage, bodySent: request.bodySent, bytes: request.bytes,
    sha256: request.digest, checkpoint: request.checkpoint, toolDigests: request.tools.map(tool => ({ sha256: tool.digest,
      bytes: tool.bytes, middleRetained: tool.middle, legacyPruned: tool.pruned })), responseStatus: request.responseStatus ?? null,
    responseBytes: request.responseBytes ?? null, responseEnd: request.responseEnd ?? null,
    usage: request.usage ?? { inspectable: false } }));
  receipt.observerEvents = observerEvents;
  receipt.actualRequests = observations.length;
  fs.writeFileSync(path.join(directory, 'native.json'), JSON.stringify(receipt, null, 2) + '\n');
  console.log(JSON.stringify(receipt));
}
