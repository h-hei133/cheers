function waitFor(predicate) {
  const found = responses.find(predicate); if (found) return Promise.resolve(found);
  return new Promise((resolve, reject) => {
    const entry = { predicate, resolve: value => { clearTimeout(deadline); waiters.delete(entry); resolve(value); } };
    const deadline = setTimeout(() => { waiters.delete(entry); reject(new Error('RPC probe response deadline.')); }, 25_000);
    waiters.add(entry);
  });
}

}

  }
}
