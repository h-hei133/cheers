function waitFor(predicate) {
  const found = responses.find(predicate); if (found) return Promise.resolve(found);
  return new Promise((resolve, reject) => {
    const entry = { predicate, resolve: value => { clearTimeout(deadline); waiters.delete(entry); resolve(value); } };
    const deadline = setTimeout(() => { waiters.delete(entry); reject(new Error('RPC probe response deadline.')); }, 25_000);
    waiters.add(entry);
  });
}
function command(type, values = {}) {
  const id = String(++number); const pending = waitFor(value => value.type === 'response' && value.id === id);
  child.stdin.write(JSON.stringify({ id, type, ...values }) + '\n'); return pending;
}
function accept(chunk) {

    let value; try { value = JSON.parse(line); } catch { continue; }
    responses.push(value); for (const entry of [...waiters]) if (entry.predicate(value)) entry.resolve(value);
  }
}
