import fs from 'node:fs';

import assert from 'node:assert/strict';
import { createHash } from 'node:crypto';

const repository = process.cwd();
const scratch = path.join(repository, '.tmp/rpc-acceptance-2026-11-16');
for (const [name, field, expected] of [['review-code.json', 'recommendation', 'APPROVE'], ['review-architecture.json', 'status', 'CLEAR']]) {
  const review = JSON.parse(fs.readFileSync(path.join(scratch, name), 'utf8'));
  assert.equal(review[field], expected);
  for (const [file, hash] of Object.entries(review.sourceHashes)) {
    assert.equal(createHash('sha256').update(fs.readFileSync(path.join(repository, file))).digest('hex'), hash, file);
  }
}
function revise(file, operation) {
  const target = path.join(repository, file);
  const before = fs.readFileSync(target, 'utf8');
  // These three documents were written before the first summary matcher failed.
  if (['.wiki/refactor/upgrade/MASTER.md', '.wiki/refactor/upgrade/tasks.md', '.wiki/refactor/upgrade/design.md'].includes(file)) {
    assert.ok(before.includes('43da5c29fee3'), file);
    return;
  }
  const after = operation(before);
  assert.notEqual(after, before, file);
  fs.writeFileSync(target, after);
}
function replace(text, pattern, value) {
  assert.ok(pattern.test(text), String(pattern));
  return text.replace(pattern, value);
}
