import fs from 'node:fs';
import path from 'node:path';
import { createHash } from 'node:crypto';
import { execFileSync } from 'node:child_process';

const scratch = '.tmp/stop-envelope-validation';
const evidence = 'docs/architecture/hooks/evidence/stop-envelope';
const hashes = JSON.parse(fs.readFileSync(`${scratch}/source-strict.json`, 'utf8'));
const sha = bytes => createHash('sha256').update(bytes).digest('hex');
const drift = Object.entries(hashes).filter(([file, hash]) => !fs.existsSync(file) || sha(fs.readFileSync(file)) !== hash);
if (drift.length) throw new Error(`Source drift: ${drift.map(([file]) => file).join(', ')}`);
const fullOutput = fs.readFileSync(`${scratch}/full-suite.txt`, 'utf8');
if (!/Test Files\s+1098 passed \(1098\)/.test(fullOutput)
  || !/Tests\s+14358 passed \| 2 skipped \(14360\)/.test(fullOutput)) throw new Error('Unexpected full suite result');
for (const [source, destination] of [
  ['full-suite.txt', 'full-suite.txt'],
  ['focused-strict.txt', 'focused-strict.txt'],
  ['build-strict.txt', 'build-strict.txt'],
  ['final-gates.txt', 'final-gates.txt'],
  ['structure-final.txt', 'structure-final.txt'],
  ['source-strict.json', 'source-files.json'],
  ['judge-output/verdict.json', 'verdict.json'],
]) fs.copyFileSync(path.join(scratch, source), path.join(evidence, destination));

const probes = fs.readFileSync(`${scratch}/focused-strict.txt`, 'utf8').split('\n')
  .filter(line => line.startsWith('{')).map(line => JSON.parse(line));
fs.writeFileSync(`${evidence}/accepted-probes.json`, `${JSON.stringify(probes, null, 2)}\n`);
const neutral = {};
for (const reason of ['already_blocked_once', 'not_configured']) {
  const probe = probes.find(item => item.probe === 'envelope-lifecycle'
    && item.finalEvent?.type === 'query.completed_unproven' && item.finalEvent.reasonCode === reason && item.projections);
  if (!probe) throw new Error(`Missing neutral projection: ${reason}`);
  neutral[reason] = probe;
}
fs.writeFileSync(`${evidence}/neutral-projections.json`, `${JSON.stringify(neutral, null, 2)}\n`);

const changed = execFileSync('git', ['diff', '--name-only'], { encoding: 'utf8' }).trim().split('\n')
  .filter(file => file.startsWith('packages/ts-shell/src/') && !file.endsWith('.test.ts'))
  .map(file => file.slice('packages/ts-shell/src/'.length));
const structureRows = fs.readFileSync(`${scratch}/structure-final.txt`, 'utf8').split('\n')
  .filter(line => changed.some(file => line.trimEnd().endsWith(` ${file}`)));
if (structureRows.some(line => /^\s*monolith\b/.test(line))) throw new Error('Changed module exceeds function ceiling');
fs.writeFileSync(`${evidence}/structure-final-touched.txt`, `${structureRows.join('\n')}\n`);

const manifest = JSON.parse(fs.readFileSync('packages/ts-shell/dist/build-manifest.json', 'utf8'));
const sortedHashes = Object.fromEntries(Object.entries(hashes).sort(([a], [b]) => a.localeCompare(b)));
const identity = {
  head: execFileSync('git', ['rev-parse', 'HEAD'], { encoding: 'utf8' }).trim(),
  sourceFiles: Object.keys(hashes).filter(file => file.startsWith('packages/ts-shell/src/')).length,
  frozenFiles: Object.keys(hashes).length,
  sourceSha256: sha(JSON.stringify(sortedHashes)),
  sourceDigestFormat: 'SHA-256 of JSON.stringify of source-files.json entries sorted by localeCompare path',
  sourceFileHashes: 'source-files.json',
  build: manifest.buildId.slice(0, 12),
  buildId: manifest.buildId,
  fullSuiteExit: 0,
  fullSuiteSourceDrift: 0,
  testFilesPassed: 1098,
  testsPassed: 14358,
  testsSkipped: 2,
  fullSuiteDurationSeconds: 1110.60,
  hashes: Object.fromEntries(['package.json', 'package-lock.json', 'packages/ts-shell/package.json'].map(file => [file, hashes[file]])),
};
fs.writeFileSync(`${evidence}/source-identity.json`, `${JSON.stringify(identity, null, 2)}\n`);
console.log(JSON.stringify({ drift: drift.length, frozenFiles: identity.frozenFiles, probes: probes.length,
  changedCallableRows: structureRows.length, build: identity.build, fullSuiteExit: identity.fullSuiteExit }));
